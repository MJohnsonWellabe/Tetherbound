# Halder r1 data, but default framing (side-on view off: it framed far),
# Tuskroot grown 1.4 -> 2.2 (grow-only; C2 re-run before ON), a semi-solid amber
# lane with amber crackle (the soft lane vanished on grass, the solid one read as
# a pad), a bigger earth strike, and depth-tested tells.

    assert s.count(a)==1,a; s=s.replace(a,b)
json.loads(s); open(p,'w').write(s)
p='data/config/combat.json'; s=open(p).read()
s=s.replace('"depth_pull_m": 0.0,','"depth_pull_m": 0.45,',1)

        "cue_lane_locked_fill_alpha": 0.38,
