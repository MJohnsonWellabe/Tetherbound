# T3 overlay (run from repo root): T2 + electric strike style on every Stormwood
# named storm row, bluer white-violet tells, larger Stormwood pilot floor.
import json,re
exec(open('ralph/reports/FIGHTS/close-1022/cand_t2.py.txt').read())
p='data/config/combat.json'; s=open(p).read()
s=s.replace('#b48cff','#9d8cff')
named=['hollows_alpha','capacitor_alpha','crown_guardian','old_rodfolk_hall_guardian','glass_field_alpha']
ELEC='"cue_impact_burst": true, "cue_burst_style": "electric", "cue_burst_arcs": 8, "cue_burst_arc_radius_m": 0.16, "cue_burst_colour": "#9d7bff", "cue_burst_foam": "#ffffff", "cue_burst_light_colour": "#d9c8ff", "cue_water_burst_light_energy": 10.0, '
def fix(m):
    key=m.group(1)
    if not any(key.startswith(n) for n in named): return m.group(0)
    body=re.sub(r'"cue_impact_burst": true, "cue_burst_colour": "#9d7bff", "cue_burst_foam": "#ffffff", "cue_burst_light_colour": "#d9c8ff", "cue_burst_arcs": 7, "cue_water_burst_light_energy": 10.0, ','',m.group(2))
    return '"%s": {%s%s'%(key,ELEC,body)
s=re.sub(r'"([a-z_]+)": \{("telegraph_colour"[^\n]*\n)',fix,s)
json.loads(s); open(p,'w').write(s)
p='data/config/foliage_camera_visibility.json'; s=open(p).read()
