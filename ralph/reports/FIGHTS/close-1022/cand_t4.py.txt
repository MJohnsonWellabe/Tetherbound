# T4 overlay (run from repo root): T3 + the leap/route lane colour and crackle on
# every Stormwood named row (capacitor_alpha_diver_leap's route line was magenta).
import json,re
exec(open('ralph/reports/FIGHTS/close-1022/cand_t3.py.txt').read())
p='data/config/combat.json'; s=open(p).read()
def fix(m):
    line=m.group(0)
    if '"cue_lane_colour"' in line: return line
    return line.replace('"telegraph_colour": "#9d8cff",','"telegraph_colour": "#9d8cff", "cue_lane_colour": "#9d8cff", "cue_lane_crackle": 0.85,')
s=re.sub(r'"[a-z_]+": \{"cue_impact_burst": true, "cue_burst_style": "electric"[^\n]*\n',fix,s)
json.loads(s); open(p,'w').write(s)
