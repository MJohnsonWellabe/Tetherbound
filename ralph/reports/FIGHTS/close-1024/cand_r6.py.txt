# Fights r6 candidate overlay (capture-only; repo root, clean tree on the r6 code). The judges'
# common r4/r5 asks on top of the shipped generic presentation: dust fills the danger footprint
# with a timing pulse, smaller right-hand panels, a fixed meter row in the compact command lane,
# and the pilot size floor in every fight. `git checkout data/config/` restores.
import json
def edit(p, pairs):
    s = open(p).read()
    for a, b in pairs:
        assert s.count(a) == 1, (p, a)
        s = s.replace(a, b)
    json.loads(s); open(p, 'w').write(s)
edit('data/config/combat.json', [
    ('"dust_alpha": 0.9,', '"dust_alpha": 0.9, "dust_fill_rows": 2, "dust_fill_reveal": 0.8, "dust_pulse": 0.45, "dust_pulse_hz": 1.5, "dust_pulse_hz_end": 4.5, "dust_max": 260,'),
])
edit('data/config/hud.json', [('"panel_scale": 1.0', '"panel_scale": 0.8')])
edit('data/config/tether_commands.json', [('"compact_rows_fixed": false', '"compact_rows_fixed": true')])
edit('data/config/camera.json', [('"pilot_min_screen_fraction": 0.0,', '"pilot_min_screen_fraction": 0.1,')])
print('r6 candidate applied')
