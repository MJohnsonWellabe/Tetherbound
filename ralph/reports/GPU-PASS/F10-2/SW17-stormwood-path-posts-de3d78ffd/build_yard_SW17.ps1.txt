# SW17 (local): Stormwood m39 junction lamps on path posts (Meshy lantern post), tb/stormwood-m39-on de3d78ffd vs its parent a933314a4.
# capture_stormwood_f10_matrix.gd, custom stands pv/pm/pd (path junctions), calm, Medium and High (queue 48 re-run with --source-commit).
# One set per stand per preset, one full frame per side.
$o = "D:\tb-gpu-runs\out"; $od = "D:\tb-gpu-runs\yard\SW17\sides"; New-Item -ItemType Directory -Force $od | Out-Null
function Side([string[]] $files, [string[]] $caps, [string] $out) {
    $s = [ordered]@{ title = ""; cols = 1; thumb_w = 1280; label_w = 4; rows = @([ordered]@{ label = ""; files = $files; captions = $caps }); out = $out }
    [IO.File]::WriteAllText("D:\tb-gpu-runs\_spec_yard_sw17.json", ($s | ConvertTo-Json -Depth 5)); & D:\tb-gpu-runs\tools\make_sheet.ps1 -SpecPath "D:\tb-gpu-runs\_spec_yard_sw17.json" | Out-Null; return $out
}
$generic = @("Placeholder or untextured materials (flat grey, checker or faceted blocks)", "Floating bodies or props (a gap between an object and the ground under it)", "Clipping (geometry passing through geometry, or the camera inside geometry)")
$lamps = "the path junction lamps hang in mid-air as glowing lanterns with nothing holding them up, rather than lanterns on posts set into the ground"
$sets = @(); $cover = @()
foreach ($pr in "Medium", "High") {
    foreach ($st in "pv", "pm", "pd") {
        $fa = "$o\SW17-cand-$pr\frames\$($st)_calm.jpg"; $fb = "$o\SW17-ctl-$pr\frames\$($st)_calm.jpg"
        if (-not ((Test-Path $fa) -and (Test-Path $fb))) { "MISSING $st $pr"; continue }
        $sa = Side @($fa) @("$st calm") "$od\SW17-$st-cand-$pr.png"; $sb = Side @($fb) @("$st calm") "$od\SW17-$st-ship-$pr.png"
        $brief = "One fixed view of a path junction in the Stormwood (a storm forest under a purple sky), calm weather, with the trainer as a 1.80 m ruler. Judge whether the junction reads as a believable, crafted waypoint in the key art's world: the lamps, the path and the dressing around it, and whether everything sits in the world."
        $sets += [ordered]@{ label = "SW17 $st $pr cand|ship"; a = $sa; b = $sb; brief = $brief; defects = @($lamps) + $generic }
        $cover += [ordered]@{ stand = $st; preset = $pr }
    }
}
$rng = New-Object System.Random 2161; $sets = @($sets | Sort-Object { $rng.Next() })
New-Item -ItemType Directory -Force "D:\tb-gpu-runs\yard\SW17\b1" | Out-Null; [IO.File]::WriteAllText("D:\tb-gpu-runs\yard\SW17\b1\candidates.json", (ConvertTo-Json -InputObject @($sets) -Depth 5))
[IO.File]::WriteAllText("D:\tb-gpu-runs\yard\SW17\coverage.json", (ConvertTo-Json -InputObject @($cover) -Depth 4))
"SW17 sets: $($sets.Count)"
