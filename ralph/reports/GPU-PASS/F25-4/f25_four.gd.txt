extends "res://tests/smoke_move_effects_library.gd"

## GPU-pass F25#4 timing wrapper (local only, never committed). Same staging and
## launch path as the smoke's four-ultimates-on-one-frame case, fired as
## TB_F25_VOLLEYS volleys (default 2) in one process: volley 1 carries first-use
## pipeline compilation, volley 2 is the warm repeat. The smoke's mid-flight PNG
## beat captures are left out here (a 1080p readback + encode is ~0.6 s and would
## be measured as a hitch); visual evidence comes from the smoke's own runs.
## Every frame logs wall interval, video memory, the encounter's particle slots,
## draw calls, primitives and the volley it belongs to.
## TB_VFX_BUDGET=off lifts every particle/mesh ceiling in-process (library and
## ultimates configs); nothing on disk changes. Vsync is disabled so the
## interval is the real frame cost.

const FOUR_SCOPE := "f35-four"
var _log: Array = []
var _last_us := 0
var _volley := 0
var _first_of_volley := false


func _initialize() -> void:
	DisplayServer.window_set_vsync_mode(DisplayServer.VSYNC_DISABLED)
	var budget_off := OS.get_environment("TB_VFX_BUDGET") == "off"
	print("F25 wrapper budget=", "off" if budget_off else "on")
	if budget_off:
		var lib: Dictionary = LIBRARY.config()
		for key: String in ["ordinary_particle_limit", "ultimate_particle_limit", "encounter_particle_cap"]:
			lib[key] = 1000000
		var ult: Dictionary = ULTIMATES.config()
		var b: Dictionary = ult.get("budget", {})
		# Ceilings only; impact_motes/trail_motes are each effect's own request.
		for key: String in ["effect_mesh_cap", "encounter_mesh_cap", "scene_mesh_cap", "encounter_particle_cap"]:
			b[key] = 1000000
	RenderingServer.frame_post_draw.connect(_on_post_draw)
	super._initialize()


func _on_post_draw() -> void:
	var now := Time.get_ticks_usec()
	if _last_us > 0:
		_log.append([now, now - _last_us, int(Performance.get_monitor(Performance.RENDER_VIDEO_MEM_USED)),
			BUDGET.used(FOUR_SCOPE), int(Performance.get_monitor(Performance.RENDER_TOTAL_DRAW_CALLS_IN_FRAME)),
			int(Performance.get_monitor(Performance.RENDER_TOTAL_PRIMITIVES_IN_FRAME)), _volley, 1 if _first_of_volley else 0])
		_first_of_volley = false
	_last_us = now


func _exercise_simultaneous(cfg: Dictionary) -> void:
	var lanes := [-4.5, -1.5, 1.5, 4.5]
	var origins: Array[Vector3] = []
	var bodies: Array[Node3D] = []
	for i in _simultaneous.size():
		var move_id := str(_simultaneous[i])
		if not _moves.has(move_id): _failures.append("Simultaneous ultimate without move " + move_id); return
		var case := {"id": move_id}
		var species := move_id.trim_prefix("ultimate_")
		if bool(_moves[move_id].get("ultimate", {}).get("unique", false)) and _species_has_model(species):
			case["attacker_species"] = species
		var from := _attacker_origin(case, move_id, float(lanes[i % lanes.size()]))
		if not from.is_finite(): _failures.append("Attacker model missing " + move_id); return
		origins.append(from)
		bodies.append(_current_attacker)
	for body: Node3D in bodies:
		body.visible = true
		body.set_physics_process(false)
	var camera := get_root().get_viewport().get_camera_3d()
	if camera != null:
		var centre := _arena.to_global(Vector3((float(_scenarios.get("stage", {}).get("attacker_x", -3.4)) + _target_x) * 0.5, 1.6, 0.0))
		camera.global_position = centre + _arena.global_basis * Vector3(-13.0, 8.0, 5.0)
		camera.look_at(centre, Vector3.UP)
	var to := _target.global_position + Vector3.UP * float(_target.call("body_height")) * 0.5
	var vs := OS.get_environment("TB_F25_VOLLEYS")
	var volleys := 2 if vs.is_empty() else maxi(1, int(vs))
	for v in volleys:
		# Settle a second between volleys so each window starts from rest.
		for k in 30: await RenderingServer.frame_post_draw
		var duration := 0.0
		var arrivals := [0]
		_volley = v + 1
		_first_of_volley = true
		for i in _simultaneous.size():
			var move_id := str(_simultaneous[i])
			var move: Dictionary = _moves[move_id]
			var signature: Dictionary = move.get("ultimate", {})
			duration = maxf(duration, float(signature.get("presentation_seconds", 2.4)))
			var binding := {"character_id": "f35-four-%d" % i, "creature_uid": "f35-four-%d-attacker" % i,
				"encounter_id": FOUR_SCOPE, "generation": 1, "action": v * 10 + i + 1}
			var spec := {"slot": "ultimate", "move_id": move_id, "action_id": "f35-four:%d" % (v * 10 + i + 1),
				"actor_binding": binding, "mastery_rank": 1, "breakthrough_count": 0,
				"ultimate": signature.duplicate(true), "vfx": move.get("vfx", {}).duplicate(true)}
			var context := {"current_actor": binding, "travel_seconds": minf(float(cfg.travel_seconds), 1.2),
				"recipient_character_id": "f35-four-%d" % i, "source_ground": _ground_point(_arena.to_local(origins[i]).x, float(lanes[i % lanes.size()])),
				"target_ground": _ground_point(_target_x, 0.0)}
			var effect: Node3D = ULTIMATES.launch(_arena, origins[i], to, spec, context)
			if effect == null: _failures.append("Simultaneous launch refused %s volley %d" % [move_id, v + 1]); continue
			effect.connect("arrived", func() -> void: arrivals[0] += 1)
			if bodies[i].has_method("play_attack"): bodies[i].call("play_attack")
		var started := Time.get_ticks_usec()
		await RenderingServer.frame_post_draw  # launch frame (logged with first_of_volley=1)
		while float(Time.get_ticks_usec() - started) / 1000000.0 < duration + 0.4:
			await RenderingServer.frame_post_draw
		if int(arrivals[0]) != _simultaneous.size(): _failures.append("Volley %d arrivals %d" % [v + 1, arrivals[0]])
		print("F25 volley %d arrivals=%d duration=%.2fs" % [v + 1, arrivals[0], duration])
		_volley = 0
	_records.append({"id": "four_simultaneous_timing", "moves": _simultaneous, "volleys": volleys})


func _finalize() -> void:
	var out := OS.get_environment("TB_F25_LOG")
	if out.is_empty():
		return
	var f := FileAccess.open(out, FileAccess.WRITE)
	f.store_line("us,dt_us,vram_bytes,budget_slots,draw_calls,primitives,volley,first_of_volley")
	for r: Array in _log:
		f.store_line("%d,%d,%d,%d,%d,%d,%d,%d" % r)
	f.close()
