param([string] $Root = "D:\tb-gpu-runs\out\F25-4", [string] $OutCsv = "D:\tb-gpu-runs\out\F25-4\f25_stats.csv")
# F25#4 per-run, per-volley stats (local tool). Each volley's launch frame (the frame that
# carries the four launches and any first-use pipeline compilation) is reported apart;
# avg/p95/p99/max cover the rest of that volley's presentation window.
$rows = @()
foreach ($d in Get-ChildItem $Root -Directory | Sort-Object Name) {
    $log = Join-Path $d.FullName "frames_log.csv"
    if (-not (Test-Path $log)) { continue }
    $c = @(Import-Csv $log)
    foreach ($v in @($c | Where-Object { [int]$_.volley -gt 0 } | ForEach-Object { [int]$_.volley } | Sort-Object -Unique)) {
        $w = @($c | Where-Object { [int]$_.volley -eq $v })
        $launch = @($w | Where-Object { $_.first_of_volley -eq '1' })
        $f = @($w | Where-Object { $_.first_of_volley -ne '1' })
        $ms = @($f | ForEach-Object { [double]$_.dt_us / 1000.0 } | Sort-Object)
        $n = $ms.Count; if ($n -eq 0) { continue }
        $avg = ($ms | Measure-Object -Average).Average
        $rows += [pscustomobject]@{ run = $d.Name; volley = $v; frames = $n
            launch_ms = $(if ($launch) { [math]::Round([double]$launch[0].dt_us / 1000.0, 1) } else { "" })
            avg_ms = [math]::Round($avg, 2); fps_avg = [math]::Round(1000 / $avg, 1)
            p95_ms = [math]::Round($ms[[int][math]::Floor(($n - 1) * 0.95)], 2); p99_ms = [math]::Round($ms[[int][math]::Floor(($n - 1) * 0.99)], 2)
            max_ms = [math]::Round($ms[-1], 2); over_33ms = @($ms | Where-Object { $_ -gt 33.33 }).Count
            vram_peak_mb = [math]::Round((($w | ForEach-Object { [double]$_.vram_bytes } | Measure-Object -Maximum).Maximum / 1MB), 0)
            particle_slots_peak = ($w | ForEach-Object { [int]$_.budget_slots } | Measure-Object -Maximum).Maximum
            draw_calls_peak = ($w | ForEach-Object { [int]$_.draw_calls } | Measure-Object -Maximum).Maximum
            primitives_peak = ($w | ForEach-Object { [int]$_.primitives } | Measure-Object -Maximum).Maximum }
    }
}
$rows | Export-Csv -NoTypeInformation -Encoding utf8 $OutCsv
$rows | Format-Table run, volley, frames, launch_ms, avg_ms, fps_avg, p95_ms, p99_ms, max_ms, over_33ms, vram_peak_mb, particle_slots_peak, draw_calls_peak -AutoSize | Out-String -Width 250
