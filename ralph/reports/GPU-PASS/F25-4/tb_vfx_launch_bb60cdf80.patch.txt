Subject: [PATCH] F25#4: keep effect material shader variants compiled across
 launches (launch stall fix)

On Forward+ a BaseMaterial3D shader variant compiles on the first get_rid() of
a material with that feature set and is freed with the last such material.
Short-lived move effects therefore recompiled the same 3-4 variants on every
launch frame: 155-350 ms for four simultaneous ultimates on Medium (GTX 1060),
cold and warm alike; Compatibility paid under 1 ms per variant.

- material_variant_pin.gd: holds one duplicate per variant for the process.
- move_effect_geometry / ultimate_geometry material(), the emissive core and
  the opacity fade's opaque->alpha switch pin their variant.
- move_effect_library.prewarm() builds the standard variants, every authored
  surface and the contact-stage shaders; move_projectile.prewarm() exposes it
  and encounter_director._ready() calls it once while the world loads.
- tests/test_material_variant_pin.gd: a second launch's materials find their
  variants pinned; prewarm is idempotent and builds every authored surface.

GPU proof (ralph/reports/GPU-PASS/F25-4/ on tb/gpu-pass): launch frame
16-30 ms cold, 10-23 ms warm; no frame over 33 ms in 24 four-ultimate volleys.

+extends RefCounted

+## F25#4 launch hitch. Forward+ compiles a BaseMaterial3D shader the first time
+## a material with a given feature set asks for its RID, and frees that shader
+## again when the last material with those features is released. Effects are
+## short-lived, so every volley re-compiled the same few variants on its
+## launch frame: 10-43 ms each, 155-350 ms for four ultimates on a GTX 1060
+## (Compatibility: under 1 ms). One duplicate per variant is held for the
+## process, so later materials of that variant reuse the compiled shader.
+## Colour, alpha value and roughness are uniforms and never make a new variant.
+static var _pins: Dictionary = {}
+static var _prewarmed := false
+
+
+static func pin(material: Material) -> void:
+	var base := material as BaseMaterial3D
+	if base == null: return
+	var key := variant_key(base)
+	if _pins.has(key): return
+	var keep := base.duplicate() as BaseMaterial3D
+	keep.get_rid()
+	_pins[key] = keep
+
+
+## The features effect materials actually vary. Two materials with the same key
+## share one compiled shader; a key that misses a feature only costs a pin.
+static func variant_key(m: BaseMaterial3D) -> String:
+	return "%d|%d|%d|%d|%d|%d|%d|%d" % [m.shading_mode, m.transparency, m.blend_mode, m.cull_mode,
+		int(m.vertex_color_use_as_albedo), int(m.no_depth_test), int(m.emission_enabled), m.billboard_mode]
+
+
+static func pinned_count() -> int:
+	return _pins.size()
+
+
+## The variants move_effect_geometry.material() and ultimate_geometry.material()
+## build (lit/unlit x opaque/alpha) plus the emissive solid core. Idempotent.
+static func prewarm() -> void:
+	if _prewarmed: return
+	_prewarmed = true
+	for lit: bool in [false, true]:
+		for alpha: bool in [false, true]:
+			for emission: bool in ([false, true] if lit else [false]):
+				var m := StandardMaterial3D.new()
+				m.shading_mode = BaseMaterial3D.SHADING_MODE_PER_PIXEL if lit else BaseMaterial3D.SHADING_MODE_UNSHADED
+				m.transparency = BaseMaterial3D.TRANSPARENCY_ALPHA if alpha else BaseMaterial3D.TRANSPARENCY_DISABLED
+				m.blend_mode = BaseMaterial3D.BLEND_MODE_MIX
+				m.cull_mode = BaseMaterial3D.CULL_DISABLED
+				m.vertex_color_use_as_albedo = true
+				m.no_depth_test = false
+				m.emission_enabled = emission
+				pin(m)

+## F25#4: Forward+ frees a BaseMaterial3D shader variant when its last material
+## is released, so short-lived effects recompiled it on every launch frame.
+## The materials of a second launch must find their variants still pinned.
+func test_effect_material_variants_stay_pinned_across_launches() -> void:
+	PIN.prewarm()
+	var prewarmed := PIN.pinned_count()
+	var first: Array = [GEOMETRY.material(Color.RED, 1.0, false), GEOMETRY.material(Color.BLUE, 0.5, true),
+		SIGNATURE.material(Color.GREEN, 1.0, true), SIGNATURE.material(Color.GREEN, 0.4, false)]
+	assert_eq(PIN.pinned_count(), prewarmed, "prewarm already covers every effect geometry material variant")
+	first.clear()
+	var second: Array = [GEOMETRY.material(Color.WHITE, 1.0, false), GEOMETRY.material(Color.BLACK, 0.4, true),
+		SIGNATURE.material(Color.YELLOW, 1.0, true), SIGNATURE.material(Color.YELLOW, 0.4, false)]
+	assert_eq(PIN.pinned_count(), prewarmed, "a second launch of the same variants adds no pins")
+	for material: StandardMaterial3D in second:
+		assert_true(PIN._pins.has(PIN.variant_key(material)), "variant %s stays pinned after the first launch is released" % PIN.variant_key(material))
+
+
+## The encounter director calls this while the world loads; every authored
+## surface must build from an empty profile without errors, twice.
+func test_library_prewarm_builds_every_authored_surface_once() -> void:
+	var library := preload("res://scripts/vfx/move_effect_library.gd")
+	library.prewarm()
+	var pinned := PIN.pinned_count()
+	assert_true(pinned >= 6, "standard lit/unlit x opaque/alpha and emissive variants are pinned")
+	library.prewarm()
+	assert_eq(PIN.pinned_count(), pinned, "prewarm is idempotent")
+	for kind: String in library.AUTHORED_KINDS:
+		assert_true(GEOMETRY.authored_material(kind, {}, Color.WHITE) is Material, kind + " builds from an empty profile")
+
+
+## Colour and alpha value are uniforms; only feature flags make a variant.
+func test_variant_key_ignores_uniforms_and_tracks_features() -> void:
+	var a := GEOMETRY.material(Color.RED, 0.3, true)
+	var b := GEOMETRY.material(Color.BLUE, 0.9, true)
+	assert_eq(PIN.variant_key(a), PIN.variant_key(b), "colour and alpha value share one variant")
+	var opaque := GEOMETRY.material(Color.RED, 1.0, true)
+	assert_ne(PIN.variant_key(opaque), PIN.variant_key(a), "opaque and alpha are different variants")
+	var emissive := opaque.duplicate() as StandardMaterial3D
+	emissive.emission_enabled = true
+	assert_ne(PIN.variant_key(emissive), PIN.variant_key(opaque), "emission is a separate variant")
+	PIN.pin(emissive)
+	assert_true(PIN._pins.has(PIN.variant_key(emissive)), "an emissive core variant can be pinned")
