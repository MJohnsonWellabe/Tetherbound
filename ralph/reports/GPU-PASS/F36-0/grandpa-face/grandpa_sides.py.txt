"""Grandpa face sides for the yardstick (local GPU-pass helper): per arm and preset, two face close-ups (front and
three-quarter, cropped around the head and upscaled) plus both full-body frames, in one strip with captions.

    python grandpa_sides.py        writes yard/grandpa/sides/grandpa-<arm>-<preset>.png and yard/grandpa/b1/candidates.json
"""
import json
from pathlib import Path

from PIL import Image, ImageDraw, ImageFont

OUT = Path("D:/tb-gpu-runs/out")
SIDES = Path("D:/tb-gpu-runs/yard/grandpa/sides")
H = 520


def head_box(img):
    """Head crop: the character's topmost non-background rows, found by scanning for pixels that differ from the corner colour."""
    rgb = img.convert("RGB")
    w, h = rgb.size
    bg = rgb.getpixel((5, 5))
    px = rgb.load()
    top = None
    for y in range(0, h // 2, 2):
        if any(sum(abs(a - b) for a, b in zip(px[x, y], bg)) > 60 for x in range(w // 3, 2 * w // 3, 3)):
            top = y
            break
    top = top or h // 5
    cols = [x for x in range(w // 3, 2 * w // 3, 2) if sum(abs(a - b) for a, b in zip(px[x, top + 40], bg)) > 60]
    cx = (min(cols) + max(cols)) // 2 if cols else w // 2
    size = 230
    return (cx - size // 2, top - 20, cx + size // 2, top - 20 + size)


def fit(img, height):
    return img.resize((int(img.width * height / img.height), height), Image.LANCZOS)


def strip(arm, preset):
    d = OUT / f"grandpa-{arm}-{preset}" / "frames"
    front, tq = Image.open(d / "02-grandpa-front.png"), Image.open(d / "02-grandpa-threequarter.png")
    tiles = [(fit(front.crop(head_box(front)), H), "face front"), (fit(tq.crop(head_box(tq)), H), "face three-quarter"),
             (fit(front.crop((560, 120, 1360, 1060)), H), "full body front"), (fit(tq.crop((560, 120, 1360, 1060)), H), "full body three-quarter")]
    width = sum(t.width for t, _ in tiles) + 8 * (len(tiles) + 1)
    sheet = Image.new("RGB", (width, H + 16), (24, 24, 24))
    draw = ImageDraw.Draw(sheet)
    try:
        font = ImageFont.truetype("segoeui.ttf", 18)
    except OSError:
        font = ImageFont.load_default()
    x = 8
    for t, cap in tiles:
        sheet.paste(t, (x, 8))
        draw.text((x + 8, 12), cap, fill=(240, 240, 240), font=font)
        x += t.width + 8
    SIDES.mkdir(parents=True, exist_ok=True)
    out = SIDES / f"grandpa-{arm}-{preset}.png"
    sheet.save(out)
    return str(out)


brief = ("One named villager or enemy-faction member: two face close-ups and a full-body view. Judge the character as the game's "
         "real look - identity, face and eyes, silhouette, palette, appeal and fit with the key art's world.")
sets = [{"label": f"Grandpa face {p} cand|ctl", "a": strip("cand", p), "b": strip("ctl", p), "brief": brief, "defects": []}
        for p in ("Medium", "High")]
Path("D:/tb-gpu-runs/yard/grandpa/b1").mkdir(parents=True, exist_ok=True)
Path("D:/tb-gpu-runs/yard/grandpa/b1/candidates.json").write_text(json.dumps(sets, indent=1), encoding="utf-8")
print([s["a"] for s in sets], [s["b"] for s in sets])
