# F36#0 naturalistic creatures on main (local): per species, Medium vs High sides; two batches of seven.
function Side([string[]] $files, [string[]] $caps, [string] $out) {
    $s = [ordered]@{ title = ""; cols = 5; thumb_w = 300; label_w = 4; rows = @([ordered]@{ label = ""; files = $files; captions = $caps }); out = $out; crop = @(620, 240, 880, 580) }
    $s | ConvertTo-Json -Depth 5 | Set-Content -Encoding utf8 "D:\tb-gpu-runs\_spec_yard_f36main.json"
    & D:\tb-gpu-runs\tools\make_sheet.ps1 -SpecPath "D:\tb-gpu-runs\_spec_yard_f36main.json" | Out-Null
    return $out
}
$od = "D:\tb-gpu-runs\yard\f36main\sides"; New-Item -ItemType Directory -Force $od | Out-Null
$brief = "One creature on a neutral stage beside a 1.80 m human ruler: standing, hit react, faint mid, faint end, and the creature in its travel role (ridden, swimming or carrying). Judge the creature's model and material as the game's real look - identity, silhouette, appeal, surface and colour, proportion and scale against the ruler. Judge poses only where they break the read."
$batches = @(
    @("ripplet", "cindercub", "terrapup", "bramblebun", "mudsnout", "trailpup", "paddlenewt"),
    @("burrowback", "veridian", "mosshell", "water_mosshell", "nightburrow", "stormtrail", "riftfrill"))
$n = 0
foreach ($b in $batches) {
    $n++; $sets = @()
    foreach ($sp in $b) {
        $role = @("ride", "swim", "fly_grip") | Where-Object { Test-Path "D:\tb-gpu-runs\out\F36-0-main\Medium\frames\$sp-$_-050.png" } | Select-Object -First 1
        $rc = @{ ride = "ridden"; swim = "swimming"; fly_grip = "carrying" }[$role]
        $poses = @(@("hit-000", "standing"), @("hit-025", "hit react"), @("faint-050", "faint mid"), @("faint-100", "faint end"), @("$role-050", $rc))
        $side = @{}
        foreach ($p in "Medium", "High") {
            $files = @($poses | ForEach-Object { "D:\tb-gpu-runs\out\F36-0-main\$p\frames\$sp-$($_[0]).png" }); $caps = @($poses | ForEach-Object { $_[1] })
            $side[$p] = Side $files $caps "$od\F36main-$sp-$p.png"
        }
        $sets += [ordered]@{ label = "F36#0 main $sp Medium|High"; a = $side["Medium"]; b = $side["High"]; brief = $brief; defects = @() }
    }
    New-Item -ItemType Directory -Force "D:\tb-gpu-runs\yard\f36main\b$n" | Out-Null
    [IO.File]::WriteAllText("D:\tb-gpu-runs\yard\f36main\b$n\candidates.json", (ConvertTo-Json -InputObject $sets -Depth 5))
    "batch $n sets: $($sets.Count)"
}
