function Pick($dir, $sp) {
    $role = @("ride", "swim", "fly_grip") | Where-Object { Test-Path "$dir\$sp-$_-050.png" } | Select-Object -First 1
    $p = @(@("hit-000", "standing"), @("hit-025", "hit react"), @("faint-050", "faint mid"), @("faint-100", "faint end"))
    if ($role) { $p += , @("$role-050", @{ ride = "ridden"; swim = "swimming"; fly_grip = "carrying" }[$role]) }
    $p = @($p | Where-Object { Test-Path "$dir\$sp-$($_[0]).png" })
    return , @(@($p | ForEach-Object { "$dir\$sp-$($_[0]).png" }), @($p | ForEach-Object { $_[1] }))
}
