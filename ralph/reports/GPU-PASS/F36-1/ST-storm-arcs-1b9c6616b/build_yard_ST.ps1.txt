$o = "D:\tb-gpu-runs\out"; $od = "D:\tb-gpu-runs\yard\ST\sides"; New-Item -ItemType Directory -Force $od | Out-Null
function Side([string[]] $files, [string[]] $caps, [string] $out) {
    $s = [ordered]@{ title = ""; cols = 5; thumb_w = 300; label_w = 4; rows = @([ordered]@{ label = ""; files = $files; captions = $caps }); out = $out; crop = @(560, 200, 1000, 640) }
    [IO.File]::WriteAllText("D:\tb-gpu-runs\_spec_yard_ST.json", ($s | ConvertTo-Json -Depth 5)); & D:\tb-gpu-runs\tools\make_sheet.ps1 -SpecPath "D:\tb-gpu-runs\_spec_yard_ST.json" | Out-Null; return $out
}
$brief = "One creature on a neutral stage beside a 1.80 m human ruler: standing, hit react, faint mid, faint end. Judge the creature's model and material as the game's real look - identity, silhouette, appeal, surface and colour, proportion and scale against the ruler. Judge poses only where they break the read."
$arcs = "the storm arcs or veins read as dull painted lines, or as a glow that floods the whole body, rather than bright arcs popping against a darker coat"
$extra = @{ "tanglevolt" = @("the coat reads lighter, more violet or washed out than the owner's dark navy Tanglevolt (owner creature style bar)", $arcs) }
$p = @(@("hit-000", "standing"), @("hit-025", "hit react"), @("faint-050", "faint mid"), @("faint-100", "faint end"))
$sets = @(); $cover = @()
foreach ($pr in "Medium", "High") {
    foreach ($sp in "tanglevolt", "staticub", "thundertunnel", "voltarach") {
        $sd = @{}; $ok = $true
        foreach ($arm in @(@("cand", "ST-cand"), @("ship", "ST-ctl"))) {
            $dir = "$o\$($arm[1])-$pr\frames"; $files = @($p | ForEach-Object { "$dir\$sp-$($_[0]).png" })
            if (@($files | Where-Object { -not (Test-Path $_) }).Count -gt 0) { $ok = $false; continue }
            $sd[$arm[0]] = Side $files @($p | ForEach-Object { $_[1] }) "$od\ST-$sp-$($arm[0])-$pr.png"
        }
        $cover += [ordered]@{ species = $sp; preset = $pr; complete = $ok }
        if ($ok) {
            $def = if ($extra.ContainsKey($sp)) { $extra[$sp] } else { , $arcs }  # comma keeps a one-item array (a bare string was spelled out per character)
            $sets += [ordered]@{ label = "ST $sp $pr cand|ship"; a = $sd["cand"]; b = $sd["ship"]; brief = $brief; defects = $def }
        }
    }
}

[IO.File]::WriteAllText("D:\tb-gpu-runs\yard\ST\coverage.json", (ConvertTo-Json -InputObject @($cover) -Depth 4))
"ST sets: $($sets.Count) in $nb batches"; $cover | ForEach-Object { "$($_.species) $($_.preset) complete=$($_.complete)" }
