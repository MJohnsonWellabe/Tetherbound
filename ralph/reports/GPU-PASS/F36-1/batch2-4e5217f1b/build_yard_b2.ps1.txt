# Creatures batch 2 (local, tb/creatures-land 4e5217f1b vs main f7b9b6a1e): one set per species, cand|ship at $env:B2_PRESET (default Medium),
# standing / hit react / faint mid / faint end + travel role where shot. Pose-recipe species take the cand side from the --candidate pass
# (B2-candrec), the rest from the plain pass. Two batches of 6, order shuffled with a fixed seed.
$o = "D:\tb-gpu-runs\out"; $od = "D:\tb-gpu-runs\yard\b2\sides"; New-Item -ItemType Directory -Force $od | Out-Null
$pr = if ($env:B2_PRESET) { $env:B2_PRESET } else { "Medium" }
function Side([string[]] $files, [string[]] $caps, [string] $out) {
    $s = [ordered]@{ title = ""; cols = 5; thumb_w = 300; label_w = 4; rows = @([ordered]@{ label = ""; files = $files; captions = $caps }); out = $out; crop = @(560, 200, 1000, 640) }
    [IO.File]::WriteAllText("D:\tb-gpu-runs\_spec_yard_b2.json", ($s | ConvertTo-Json -Depth 5)); & D:\tb-gpu-runs\tools\make_sheet.ps1 -SpecPath "D:\tb-gpu-runs\_spec_yard_b2.json" | Out-Null; return $out
}
$brief = "One creature on a neutral stage beside a 1.80 m human ruler: standing, hit react, faint mid, faint end, and the creature in its travel role (ridden, swimming or carrying) where it has one. Judge the creature's model and material as the game's real look - identity, silhouette, appeal, surface and colour, proportion and scale against the ruler. Judge poses only where they break the read."
$recipe = "mirejaw,water_mirejaw,riverdrake,water_riverdrake,mosshock".Split(",")
$species = "mudsnout,trailpup,cindercub,nightburrow,riverdrake,water_riverdrake,mirejaw,water_mirejaw,mosshock,riftfrill,paddlenewt,stormtrail".Split(",")
$sets = @(); $cover = @()
foreach ($sp in $species) {
    $sd = @{}; $n = @{}
    foreach ($arm in "cand", "ship") {
        $tag = if ($arm -eq "cand" -and $recipe -contains $sp) { "B2-candrec" } else { "B2-$arm" }
        $dir = "$o\$tag-$pr\frames"
        $role = @("ride", "swim", "fly_grip") | Where-Object { Test-Path "$dir\$sp-$_-050.png" } | Select-Object -First 1
        $p = @(@("hit-000", "standing"), @("hit-025", "hit react"), @("faint-050", "faint mid"), @("faint-100", "faint end"))
        if ($role) { $p += , @("$role-050", @{ ride = "ridden"; swim = "swimming"; fly_grip = "carrying" }[$role]) }
        $files = @($p | ForEach-Object { "$dir\$sp-$($_[0]).png" } | Where-Object { Test-Path $_ })
        $caps = @($p | Where-Object { Test-Path "$dir\$sp-$($_[0]).png" } | ForEach-Object { $_[1] })
        $n[$arm] = $files.Count
        if ($files.Count -gt 0) { $sd[$arm] = Side $files $caps "$od\$sp-$arm-$pr.png" }
    }
    $cover += [ordered]@{ species = $sp; preset = $pr; cand = $n["cand"]; ship = $n["ship"] }
    $def = @()
    if ($sp -match "mirejaw|riverdrake") { $def += "frills, fins or crest in dark red or oxblood (a colour reserved for the enemy faction)" }
    if ($sp -eq "mosshock") { $def += "a yellow-black or lime-and-black camouflage blotch pattern that breaks up the silhouette and hides the face" }
    if ($sd["cand"] -and $sd["ship"]) { $sets += [ordered]@{ label = "B2 $sp $pr cand|ship"; a = $sd["cand"]; b = $sd["ship"]; brief = $brief; defects = $def } }
}
$rng = New-Object System.Random 2043; $sets = @($sets | Sort-Object { $rng.Next() })
$nb = 0; for ($i = 0; $i -lt $sets.Count; $i += 6) { $nb++; $chunk = $sets[$i..([math]::Min($i + 5, $sets.Count - 1))]; New-Item -ItemType Directory -Force "D:\tb-gpu-runs\yard\b2\b$nb" | Out-Null; [IO.File]::WriteAllText("D:\tb-gpu-runs\yard\b2\b$nb\candidates.json", (ConvertTo-Json -InputObject @($chunk) -Depth 5)) }
[IO.File]::WriteAllText("D:\tb-gpu-runs\yard\b2\coverage.json", (ConvertTo-Json -InputObject @($cover) -Depth 4))
"b2 sets: $($sets.Count) in $nb batches; missing: " + (($cover | Where-Object { -not $_.cand -or -not $_.ship } | ForEach-Object { $_.species }) -join ",")
