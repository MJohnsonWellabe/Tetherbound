# F36#1 held shove hit on main (local): per species, recipe (--candidate) vs shipping at Medium; three batches of five.
function Side([string[]] $files, [string[]] $caps, [string] $out) {
    $s = [ordered]@{ title = ""; cols = 5; thumb_w = 300; label_w = 4; rows = @([ordered]@{ label = ""; files = $files; captions = $caps }); out = $out; crop = @(620, 240, 880, 580) }
    $s | ConvertTo-Json -Depth 5 | Set-Content -Encoding utf8 "D:\tb-gpu-runs\_spec_yard_f36_1.json"
    & D:\tb-gpu-runs\tools\make_sheet.ps1 -SpecPath "D:\tb-gpu-runs\_spec_yard_f36_1.json" | Out-Null
    return $out
}
$od = "D:\tb-gpu-runs\yard\f36_1\sides"; New-Item -ItemType Directory -Force $od | Out-Null
$brief = "One creature on a neutral stage beside a 1.80 m human ruler: standing, hit react, hit recover, faint mid, faint end. Judge the creature's model and material as the game's real look - identity, silhouette, appeal, surface and colour, proportion and scale against the ruler. Judge poses only where they break the read."
$poses = @(@("hit-000", "standing"), @("hit-025", "hit react"), @("hit-075", "hit recover"), @("faint-050", "faint mid"), @("faint-100", "faint end"))
$batches = @(@("tanglevolt", "craghorn", "cloudfang", "breezetail", "stormcapra"), @("stormraven", "tidecoil", "frostclaw", "solmane", "aquaryn"), @("cliffspike", "thundertunnel", "pebbik", "voltwig", "galecrest"))
$n = 0
foreach ($b in $batches) {
    $n++; $sets = @()
    foreach ($sp in $b) {
        $side = @{}
        foreach ($arm in "candidate", "shipping") {
            $files = @($poses | ForEach-Object { "D:\tb-gpu-runs\out\F36-1-main\Medium-$arm\frames\$sp-$($_[0]).png" }); $caps = @($poses | ForEach-Object { $_[1] })
            $side[$arm] = Side $files $caps "$od\F36-1-$sp-$arm.png"
        }
        $sets += [ordered]@{ label = "F36#1 main $sp recipe|shipping Medium"; a = $side["candidate"]; b = $side["shipping"]; brief = $brief; defects = @() }
    }
    New-Item -ItemType Directory -Force "D:\tb-gpu-runs\yard\f36_1\b$n" | Out-Null
    [IO.File]::WriteAllText("D:\tb-gpu-runs\yard\f36_1\b$n\candidates.json", (ConvertTo-Json -InputObject $sets -Depth 5))
    "batch $n sets: $($sets.Count)"
}
