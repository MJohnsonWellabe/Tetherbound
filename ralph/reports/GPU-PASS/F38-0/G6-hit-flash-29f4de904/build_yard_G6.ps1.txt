function Side([string[]] $files, [string[]] $caps, [string] $out) {
    $s = [ordered]@{ title = ""; cols = 4; thumb_w = 470; label_w = 4; rows = @([ordered]@{ label = ""; files = $files; captions = $caps }); out = $out; crop = @(300, 150, 1320, 680) }

}
$brief = "A real Meadows trainer fight seen through the production camera with ordinary attack and dodge inputs: 16 consecutive frames 0.15 s apart (read left to right, top to bottom). Judge whether the creatures read as alive in combat - in particular whether a creature that is struck visibly shows the hit on its body (a flash, tint or recoil) at the moment of impact - and whether the fight reads clearly on screen."
$sets = @()
foreach ($pr in "Medium", "High") {
    foreach ($tr in "south_bridge_grunt", "warden_aldis") {
        foreach ($half in @(@(1, 16), @(17, 32))) {
            $ids = @($half[0]..$half[1] | ForEach-Object { "{0}-{1:D2}" -f $tr, $_ })
            $sd = @{}
            foreach ($arm in "cand", "ship") {
                $src = @{ cand = "G6-cand"; ship = "G6-ship" }[$arm]; $files = @($ids | ForEach-Object { "$o\$src-$pr\frames\$_.png" } | Where-Object { Test-Path $_ })
                $sd[$arm] = Side $files @($files | ForEach-Object { [IO.Path]::GetFileNameWithoutExtension($_) -replace "^$tr-", "f" }) "$od\G6-$tr-$($half[0])-$arm-$pr.png"
            }
            $sets += [ordered]@{ label = "G6 $tr f$($half[0])-$($half[1]) $pr cand|ship"; a = $sd["cand"]; b = $sd["ship"]; brief = $brief; defects = @("P2-016: a struck creature shows no visible hit on its body (no flash, tint or recoil) at the moment of impact") }
        }
    }
}
