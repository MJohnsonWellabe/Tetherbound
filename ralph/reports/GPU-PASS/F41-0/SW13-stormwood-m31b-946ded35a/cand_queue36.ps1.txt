function Import($proj, $tag) {
    $env:APPDATA = "D:\tb-gpu-runs\appdata-Medium"
    $p = Start-Process -FilePath $godot -ArgumentList "--headless --path $proj --import" -NoNewWindow -PassThru -RedirectStandardOutput "$out\import-$tag.log" -RedirectStandardError "$out\import-$tag.err.log"
    $null = $p.Handle; if (-not $p.WaitForExit(5400000)) { Stop-Process -Id $p.Id -Force; "import $tag killed" }; "import $tag exit=$($p.ExitCode)"
}
function CheckoutCrA($sha) { for ($i = 0; $i -lt 3; $i++) { while (Get-Process Godot* -ErrorAction SilentlyContinue) { Start-Sleep 3 }; $e = git -C $crA checkout -q -f --detach $sha 2>&1; $h = (git -C $crA rev-parse HEAD 2>$null).Trim(); if ($h.StartsWith($sha)) { "crA at $($h.Substring(0,11))"; return }; "checkout $sha attempt $($i+1) failed: $e"; Start-Sleep 20 }; "CHECKOUT FAILED: crA wanted $sha, at $h; stopping queue"; exit }

function Gate($step) { if (Test-Path "D:\tb-gpu-runs\STOP_QUEUE") { "queue stopped at gate before $step"; Remove-Item "D:\tb-gpu-runs\STOP_QUEUE"; exit } }
