# Items SW16 + SW18 (local), judged in one run. F41 catalogue, clean tree at each arm's SHA.
# SW16: Ashfoot m36b seated + firelit + storm view (clear glass), tb/stormwood-m36b-on 0a633e222 (on) vs merged lane 9b28ee9dd (off).
#   ashfoot_shelter_inside in all four phases; ashfoot_shelter_outside as a no-change check (watch it: the window is now clear glass).
# SW18: Crown Overlook vista m41 (deck + Hollow Crown silhouette), tb/stormwood-m41-on aa74c9978 (on) vs parent 5135aceb2 (off).
#   glowmoss_hollows_crown_overlook, three views facing the Crown at heading 37: one set for calm, one for fading.
$od = "D:\tb-gpu-runs\yard\SW1618\sides"; New-Item -ItemType Directory -Force $od | Out-Null
function Side([string[]] $files, [string[]] $caps, [string] $out, [int] $cols = 3, [int] $tw = 420) {
    $s = [ordered]@{ title = ""; cols = $cols; thumb_w = $tw; label_w = 4; rows = @([ordered]@{ label = ""; files = $files; captions = $caps }); out = $out }
    [IO.File]::WriteAllText("D:\tb-gpu-runs\_spec_yard_SW1618.json", ($s | ConvertTo-Json -Depth 5))
    & D:\tb-gpu-runs\tools\make_sheet.ps1 -SpecPath "D:\tb-gpu-runs\_spec_yard_SW1618.json" | Out-Null
    return $out
}
$generic = @(
    "Flat magenta slabs or rings on the ground",
    "Placeholder or untextured materials (flat grey, checker or faceted blocks)",
    "Floating bodies or props (a gap between an object and the ground under it)",
    "Clipping (geometry passing through geometry, or the camera inside geometry)")
$world = @(
    "F41 gap 1: a repeated ground carpet with weak soil, moss and rock transitions",
    "F41 gap 3: rigid glass slabs or white strips and smooth cyan cones that do not sit in the terrain")
$standExtra = @{
    "ashfoot_shelter_inside" = @("the shelter interior reads as a bare plaster box with no Stormwood lining, hearth or lived-in dressing",
        "the storm window reads as a glowing slab standing off the wall, and its light does not change with the storm phase")
    "glowmoss_hollows_crown_overlook" = @("the overlook has no readable vista or landmark: no Hollow Crown silhouette and no sense of height or distance")
}
$brief = "Fixed production-camera views of one Stormwood location ({0}) through a storm's calm, building, break and fading phases, with the 1.80 m trainer as a ruler. Judge whether it reads as the key art's Stormwood: a storm-lit purple forest realm with readable landmarks, convincing ground and light and weather that belong to the world."
function Frames($root, $pr) { $h = @{}; Get-ChildItem "$root-$pr\frames" -Filter *.jpg -ErrorAction SilentlyContinue | ForEach-Object { $h[$_.BaseName] = $_.FullName }; return $h }
# item, cand root, base root, stand, groups (suffix, id regex), columns
$plan = @(
    @("SW16", "D:\tb-gpu-runs\out\SW16-cand", "D:\tb-gpu-runs\out\SW16-ctl", "ashfoot_shelter_inside", @(, @("", ".")), 2),
    @("SW16", "D:\tb-gpu-runs\out\SW16-cand", "D:\tb-gpu-runs\out\SW16-ctl", "ashfoot_shelter_outside", @(, @("", ".")), 2),
    @("SW18", "D:\tb-gpu-runs\out\SW18-cand", "D:\tb-gpu-runs\out\SW18-ctl", "glowmoss_hollows_crown_overlook", @(@("-calm", "_calm$"), @("-fading", "_fading$")), 3))
$order = @{ calm = 0; building = 1; break = 2; fading = 3 }
$sets = @(); $cover = @()
foreach ($pr in "Medium", "High") {
    foreach ($p in $plan) {
        $item, $candRoot, $baseRoot, $st, $groups, $cols = $p
        $c = Frames $candRoot $pr; $b = Frames $baseRoot $pr
        $ids = @($c.Keys | Where-Object { $b.ContainsKey($_) -and ($_ -replace '_(v\d+_)?(calm|building|break|fading)$', '') -eq $st })
        foreach ($grp in $groups) {
            $sfx = $grp[0]
            $pick = @($ids | Where-Object { $_ -match $grp[1] } | Sort-Object { $ph = ($_ -split '_')[-1]; "$($order[$ph])-$($_ -replace '_[a-z]+$', '')" })
            if ($pick.Count -eq 0) { "MISSING $item $st$sfx $pr"; continue }
            $caps = @($pick | ForEach-Object { ($_ -replace "^$st" + "_", '') -replace '_', ' ' })
            $sa = Side @($pick | ForEach-Object { $c[$_] }) $caps "$od\$item-$st$sfx-cand-$pr.png" $cols
            $sb = Side @($pick | ForEach-Object { $b[$_] }) $caps "$od\$item-$st$sfx-ship-$pr.png" $cols
            $def = @($generic) + $world; if ($standExtra.ContainsKey($st)) { $def += $standExtra[$st] }
            $sets += [ordered]@{ label = "$item $st$sfx $pr cand|ship"; a = $sa; b = $sb; brief = ($brief -f ($st -replace '_', ' ')); defects = $def }
            $cover += [ordered]@{ item = $item; stand = "$st$sfx"; preset = $pr; judged = $pick }
        }
    }
}
New-Item -ItemType Directory -Force "D:\tb-gpu-runs\yard\SW1618" | Out-Null
[IO.File]::WriteAllText("D:\tb-gpu-runs\yard\SW1618\all.json", (ConvertTo-Json -InputObject @($sets) -Depth 5))
[IO.File]::WriteAllText("D:\tb-gpu-runs\yard\SW1618\coverage.json", (ConvertTo-Json -InputObject $cover -Depth 5))
"SW16+SW18 sets: $($sets.Count)"; $cover | ForEach-Object { "$($_.item) $($_.stand) $($_.preset) frames=$($_.judged.Count)" }
