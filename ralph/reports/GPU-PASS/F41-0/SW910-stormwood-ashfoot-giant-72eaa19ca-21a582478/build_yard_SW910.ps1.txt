# Items SW9 + SW10 (local), judged in one run. F41 catalogue, clean tree at each arm's SHA.
# SW9: Ashfoot r25 on the merged head, tb/stormwood-m25-on 72eaa19ca (on) vs tb/stormwood 040903b16 (off); out\SW9-cand / SW9-ctl.
#   ashfoot_shelter_inside in all four phases (the pane changes by phase and with the surge flash); ashfoot_shelter_outside as a no-change check.
# SW10: Fallen Giant hero m28b, tb/stormwood-m28b-on 21a582478 (on) vs a7048bc4 (off); out\SW9-cand10 / SW9-ctl10.
#   deepwood_the_fallen_giant, all six views, one set for calm and one for fading (as SW7's Stormheart).
$od = "D:\tb-gpu-runs\yard\SW910\sides"; New-Item -ItemType Directory -Force $od | Out-Null
function Side([string[]] $files, [string[]] $caps, [string] $out, [int] $cols = 3, [int] $tw = 420) {
    $s = [ordered]@{ title = ""; cols = $cols; thumb_w = $tw; label_w = 4; rows = @([ordered]@{ label = ""; files = $files; captions = $caps }); out = $out }
    [IO.File]::WriteAllText("D:\tb-gpu-runs\_spec_yard_SW910.json", ($s | ConvertTo-Json -Depth 5))
    & D:\tb-gpu-runs\tools\make_sheet.ps1 -SpecPath "D:\tb-gpu-runs\_spec_yard_SW910.json" | Out-Null
    return $out
}
$generic = @(
    "Flat magenta slabs or rings on the ground",
    "Placeholder or untextured materials (flat grey, checker or faceted blocks)",
    "Floating bodies or props (a gap between an object and the ground under it)",
    "Clipping (geometry passing through geometry, or the camera inside geometry)")
$world = @(
    "F41 gap 1: a repeated ground carpet with weak soil, moss and rock transitions",
    "F41 gap 3: rigid glass slabs or white strips and smooth cyan cones that do not sit in the terrain")
$standExtra = @{
    "ashfoot_shelter_inside" = @("the shelter interior reads as a bare plaster box with no Stormwood lining, hearth or lived-in dressing")
    "deepwood_the_fallen_giant" = @("the fallen giant is missing from view or reads as a plain primitive log rather than a hero landmark")
}
$brief = "Fixed production-camera views of one Stormwood location ({0}) through a storm's calm, building, break and fading phases, with the 1.80 m trainer as a ruler. Judge whether it reads as the key art's Stormwood: a storm-lit purple forest realm with readable landmarks, convincing ground and light and weather that belong to the world."
function Frames($root, $pr) { $h = @{}; Get-ChildItem "$root-$pr\frames" -Filter *.jpg -ErrorAction SilentlyContinue | ForEach-Object { $h[$_.BaseName] = $_.FullName }; return $h }
# item, cand root, base root, stand, groups (suffix, id filter), columns
$plan = @(
    @("SW9", "D:\tb-gpu-runs\out\SW9-cand", "D:\tb-gpu-runs\out\SW9-ctl", "ashfoot_shelter_inside", @(, @("", "*")), 2),
    @("SW9", "D:\tb-gpu-runs\out\SW9-cand", "D:\tb-gpu-runs\out\SW9-ctl", "ashfoot_shelter_outside", @(, @("", "*")), 2),
    @("SW10", "D:\tb-gpu-runs\out\SW9-cand10", "D:\tb-gpu-runs\out\SW9-ctl10", "deepwood_the_fallen_giant", @(@("-calm", "*_calm"), @("-fading", "*_fading")), 3))
$sets = @(); $cover = @()
foreach ($pr in "Medium", "High") {
    foreach ($p in $plan) {
        $item, $candRoot, $baseRoot, $st, $groups, $cols = $p
        $c = Frames $candRoot $pr; $b = Frames $baseRoot $pr
        $ids = @($c.Keys | Where-Object { $b.ContainsKey($_) -and ($_ -replace '_(v\d+_)?(calm|building|break|fading)$', '') -eq $st })
        foreach ($grp in $groups) {
            $sfx = $grp[0]
            $pick = @($ids | Where-Object { $_ -like $grp[1] } | Sort-Object { $o = @{ calm = 0; building = 1; break = 2; fading = 3 }; $ph = ($_ -split '_')[-1]; "$($_ -replace '_[a-z]+$', '')-$($o[$ph])" })
            if ($pick.Count -eq 0) { "MISSING $item $st$sfx $pr"; continue }
            $caps = @($pick | ForEach-Object { ($_ -replace "^$st" + "_", '') -replace '_', ' ' })
            $sa = Side @($pick | ForEach-Object { $c[$_] }) $caps "$od\$item-$st$sfx-cand-$pr.png" $cols
            $sb = Side @($pick | ForEach-Object { $b[$_] }) $caps "$od\$item-$st$sfx-ship-$pr.png" $cols
            $def = @($generic) + $world; if ($standExtra.ContainsKey($st)) { $def += $standExtra[$st] }
            $sets += [ordered]@{ label = "$item $st$sfx $pr cand|ship"; a = $sa; b = $sb; brief = ($brief -f ($st -replace '_', ' ')); defects = $def }
            $cover += [ordered]@{ item = $item; stand = "$st$sfx"; preset = $pr; judged = $pick }
        }
    }
}
$rng = New-Object System.Random 2063; $sets = @($sets | Sort-Object { $rng.Next() })
$nb = [math]::Ceiling($sets.Count / 5); $buckets = @(for ($k = 0; $k -lt $nb; $k++) { , @() })
for ($i = 0; $i -lt $sets.Count; $i++) { $buckets[$i % $nb] += , $sets[$i] }
for ($k = 0; $k -lt $nb; $k++) { $n = $k + 1; New-Item -ItemType Directory -Force "D:\tb-gpu-runs\yard\SW910\b$n" | Out-Null; [IO.File]::WriteAllText("D:\tb-gpu-runs\yard\SW910\b$n\candidates.json", (ConvertTo-Json -InputObject @($buckets[$k]) -Depth 5)) }
[IO.File]::WriteAllText("D:\tb-gpu-runs\yard\SW910\coverage.json", (ConvertTo-Json -InputObject $cover -Depth 5))
"SW9+SW10 sets: $($sets.Count) in $nb batches"; $cover | ForEach-Object { "$($_.item) $($_.stand) $($_.preset) frames=$($_.judged.Count)" }
