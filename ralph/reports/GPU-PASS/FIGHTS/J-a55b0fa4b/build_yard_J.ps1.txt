$def = @(
    "The player's own creature is translucent, ghosted or too small to find in the fight",
    "A stagger is announced by a label under the wrong creature's panel, or with no change in the creature's body",

$brief = "Frames from one real trainer fight through the production camera, with the player's creature fighting the opponent's. Judge whether the fight reads on screen - both fighters visible, telegraphs and impacts readable, the player's own creature clearly present - and whether it looks like an event that belongs to this realm of the key art."
function Pick($dir, $fight) {
    $all = @(Get-ChildItem $dir -Recurse -Filter *.png | Sort-Object Name)
    switch ($fight) {
        "warden" { return @($all | Where-Object { $_.Name -match 't01-stand-mid|t02-stand-mid|-04\.png$|-08\.png$' }) }
        "riverwatch" { return @($all | Where-Object { $_.Name -match 't01-[a-z]+-mid|t02-[a-z]+-mid|-04\.png$|-08\.png$' }) }
        "stormwood" { return @($all | Where-Object { $_.Name -match 'tell1-b-mid|tell2-b-mid|tell1-e-impact|tell3-c-late' }) }
        "veyra" { $l = @($all | Where-Object { $_.Name -like 'live-*' }); if ($l.Count -lt 4) { return $l }; $n = $l.Count; return @($l[[int]($n*0.15)], $l[[int]($n*0.4)], $l[[int]($n*0.65)], $l[[int]($n*0.9)]) }
        "venn" { return @($all | Where-Object { $_.Name -match 'tell-start-000\.6|t-006\.0|t-012\.0|t-024\.0' } | Select-Object -First 4) }
    }
}
