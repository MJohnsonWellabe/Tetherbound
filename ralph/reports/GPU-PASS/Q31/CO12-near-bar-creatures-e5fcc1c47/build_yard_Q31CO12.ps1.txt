$o = "D:\tb-gpu-runs\out"; $od = "D:\tb-gpu-runs\yard\Q31CO12\sides"; New-Item -ItemType Directory -Force $od | Out-Null
function Side([string[]] $files, [string[]] $caps, [string] $out) {
    $s = [ordered]@{ title = ""; cols = 5; thumb_w = 300; label_w = 4; rows = @([ordered]@{ label = ""; files = $files; captions = $caps }); out = $out; crop = @(560, 200, 1000, 640) }
    [IO.File]::WriteAllText("D:\tb-gpu-runs\_spec_yard_q31co12.json", ($s | ConvertTo-Json -Depth 5)); & D:\tb-gpu-runs\tools\make_sheet.ps1 -SpecPath "D:\tb-gpu-runs\_spec_yard_q31co12.json" | Out-Null; return $out
}
$brief = "One creature on a neutral stage beside a 1.80 m human ruler: standing, hit react, faint mid, faint end, and the creature in its travel role (ridden, swimming or carrying) where it has one. Judge the creature's model and material as the game's real look - identity, silhouette, appeal, surface and colour, proportion and scale against the ruler. Judge poses only where they break the read."
$float = "the fainted body floats above the stage, with a gap between the body and its shadow"
$flinch = "the hit react barely differs from the standing pose, or the flinch lifts the body off the ground"
$def = @{
    cliffspike = @($flinch, $float); voltwig = @($flinch, $float)
    nightburrow = @("the aspect flame stays lit on the fainted body, so the faint does not read as knocked out", $float)
    thundertunnel = @("the fur reads near-black and muddy, so the body's form and markings are lost", $float) }
function Pick($dir, $sp) {
    $role = @("ride", "swim", "fly_grip") | Where-Object { Test-Path "$dir\$sp-$_-050.png" } | Select-Object -First 1
    $p = @(@("hit-000", "standing"), @("hit-025", "hit react"), @("faint-050", "faint mid"), @("faint-100", "faint end"))
    if ($role) { $p += , @("$role-050", @{ ride = "ridden"; swim = "swimming"; fly_grip = "carrying" }[$role]) }
    $p = @($p | Where-Object { Test-Path "$dir\$sp-$($_[0]).png" })
    return , @(@($p | ForEach-Object { "$dir\$sp-$($_[0]).png" }), @($p | ForEach-Object { $_[1] }))
}
$sets = @(); $cover = @()
foreach ($pr in "Medium", "High") {
    foreach ($sp in "cliffspike", "voltwig", "nightburrow", "thundertunnel") {
        $a = Pick "$o\CO12-cand-$pr-$sp\frames" $sp; $b = Pick "$o\CO12-ship-$pr-$sp\frames" $sp
        if ($a[0].Count -eq 0 -or $a[0].Count -ne $b[0].Count) { "MISSING/UNEVEN $sp $pr cand=$($a[0].Count) ship=$($b[0].Count)"; continue }
        $sa = Side $a[0] $a[1] "$od\CO12-$sp-cand-$pr.png"; $sb = Side $b[0] $b[1] "$od\CO12-$sp-ship-$pr.png"
        $sets += [ordered]@{ label = "CO12 $sp $pr cand|ship"; a = $sa; b = $sb; brief = $brief; defects = $def[$sp] }
        $cover += [ordered]@{ species = $sp; preset = $pr; poses = $a[1] }
    }
}
$rng = New-Object System.Random 3102; $sets = @($sets | Sort-Object { $rng.Next() })
$nb = [math]::Max(1, [math]::Ceiling($sets.Count / 4)); $buckets = @(for ($k = 0; $k -lt $nb; $k++) { , @() })
for ($i = 0; $i -lt $sets.Count; $i++) { $buckets[$i % $nb] += , $sets[$i] }
for ($k = 0; $k -lt $nb; $k++) { New-Item -ItemType Directory -Force "D:\tb-gpu-runs\yard\Q31CO12\b$($k+1)" | Out-Null; [IO.File]::WriteAllText("D:\tb-gpu-runs\yard\Q31CO12\b$($k+1)\candidates.json", (ConvertTo-Json -InputObject @($buckets[$k]) -Depth 5)) }
[IO.File]::WriteAllText("D:\tb-gpu-runs\yard\Q31CO12\coverage.json", (ConvertTo-Json -InputObject @($cover) -Depth 4))
"Q31CO12 sets: $($sets.Count) in $nb batches"; $cover | ForEach-Object { "$($_.species) $($_.preset) $($_.poses -join ',')" }
