# HALL-1 (F17#6 GPU swing): the F17-6 camera set in the three five-view groups of every earlier round (approach-nave, nave-sides,
# shrine), with the 24 Oct GPU pass's brief and recorded defects. Frames: out\HALL1-<arm>-<preset>\frames.
#   -One main          one-sided sets (the same strip on both sides): the baseline on current main
#   -Cand cand -Ctl main   two-sided sets, candidate vs main
param([string] $Tag = "HALL1", [string] $One = "", [string] $Cand = "", [string] $Ctl = "", [string[]] $Presets = @("High"))
$od = "D:\tb-gpu-runs\yard\$Tag\sides"; New-Item -ItemType Directory -Force $od | Out-Null
$groups = [ordered]@{
    "approach-nave" = @("hall-approach_day_clear", "hall-approach_night_clear", "nave-forward_day_clear", "nave-forward_night_clear", "nave-forward_day_rain")
    "nave-sides"    = @("nave-arches-left_day_clear", "nave-arches-left_night_clear", "nave-arches-right_day_clear", "nave-arches-right_night_clear", "nave-reverse_day_clear")
    "shrine"        = @("shrine-pedestals_day_clear", "shrine-pedestals_night_clear", "shrine-pedestals-reverse_day_clear", "shrine-pedestals-reverse_night_clear", "nave-reverse_night_clear")
}
$defects = @(
    "The Hall reads as a bare courtyard under a cold blue-grey grade instead of a warm, furnished great hall",
    "Structural posts sit in the camera's near field and frame or block the views",
    "The shrine end has no warm focal point that draws the eye")
$brief = "Fixed views of the Crossing Hall in a Meadows village (outside approach, nave, side arches, shrine end) by day and night, one in rain, with the 1.80 m trainer as a ruler. Judge whether the Hall reads as a warm, lived-in great hall that belongs in the key art's world."
function Side($arm, $p, $g) {
    $files = @($groups[$g] | ForEach-Object { "D:\tb-gpu-runs\out\HALL1-$arm-$p\frames\$_.png" })
    $miss = @($files | Where-Object { -not (Test-Path $_) }); if ($miss.Count) { "MISSING $($miss -join ', ')" | Write-Host; return $null }
    $caps = @($groups[$g] | ForEach-Object { ($_ -replace '_clear$', '') -replace '_', ' ' })
    $out = "$od\HALL1-$g-$arm-$p.png"
    $s = [ordered]@{ title = ""; cols = 3; thumb_w = 420; label_w = 4; rows = @([ordered]@{ label = ""; files = $files; captions = $caps }); out = $out }
    [IO.File]::WriteAllText("D:\tb-gpu-runs\_spec_yard_hall1.json", ($s | ConvertTo-Json -Depth 5)); & D:\tb-gpu-runs\tools\make_sheet.ps1 -SpecPath "D:\tb-gpu-runs\_spec_yard_hall1.json" | Out-Null
    return $out
}
$sets = @()
foreach ($p in $Presets) {
    foreach ($g in $groups.Keys) {
        if ($One) {
            $s = Side $One $p $g; if ($s) { $sets += [ordered]@{ label = "HALL1 $g $p one-sided"; a = $s; b = $s; brief = $brief; defects = $defects } }
        } else {
            $a = Side $Cand $p $g; $b = Side $Ctl $p $g
            if ($a -and $b) { $sets += [ordered]@{ label = "HALL1 $g $p cand|ship"; a = $a; b = $b; brief = $brief; defects = $defects } }
        }
    }
}
New-Item -ItemType Directory -Force "D:\tb-gpu-runs\yard\$Tag\b1" | Out-Null
[IO.File]::WriteAllText("D:\tb-gpu-runs\yard\$Tag\b1\candidates.json", (ConvertTo-Json -InputObject @($sets) -Depth 5))
"$Tag sets: $($sets.Count)"
