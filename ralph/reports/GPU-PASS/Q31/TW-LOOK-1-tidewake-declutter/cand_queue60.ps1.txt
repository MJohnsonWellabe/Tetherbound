function EnsureImported($sha, $tag) {
    CheckoutCrA $sha
    if ((Test-Path $mark) -and ((Get-Content $mark -Raw).Trim() -eq $sha)) { "  crA already imported at $($sha.Substring(0,9))" }
    else { Import $crA $tag; Set-Content -Path $mark -Value $sha -Encoding ascii }
}
