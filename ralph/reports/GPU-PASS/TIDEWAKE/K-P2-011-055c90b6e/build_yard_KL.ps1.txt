function Side([string[]] $files, [string[]] $caps, [string] $out) {
    $s = [ordered]@{ title = ""; cols = 1; thumb_w = 1100; label_w = 4; rows = @([ordered]@{ label = ""; files = $files; captions = $caps }); out = $out }
    [IO.File]::WriteAllText("D:\tb-gpu-runs\_spec_yard_KL.json", ($s | ConvertTo-Json -Depth 5)); & D:\tb-gpu-runs\tools\make_sheet.ps1 -SpecPath "D:\tb-gpu-runs\_spec_yard_KL.json" | Out-Null; return $out
}
