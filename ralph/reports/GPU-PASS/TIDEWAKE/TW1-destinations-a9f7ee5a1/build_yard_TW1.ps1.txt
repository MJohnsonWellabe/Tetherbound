# TW1 (local): Tidewake P2-011 destinations on a9f7ee5a1, whole candidate stack (destination_landmarks s1+s3+s4+s5) vs no sections.
# capture_tidewake_f13_5.gd, camera set P2-011-destinations.json, views pumps-shore and jetty, noon clear, Medium and High.
# One set per view per preset, one full frame per side (HUD in frame on both arms).
$o = "D:\tb-gpu-runs\out"; $od = "D:\tb-gpu-runs\yard\TW1\sides"; New-Item -ItemType Directory -Force $od | Out-Null
function Side([string[]] $files, [string[]] $caps, [string] $out) {
    $s = [ordered]@{ title = ""; cols = 1; thumb_w = 1280; label_w = 4; rows = @([ordered]@{ label = ""; files = $files; captions = $caps }); out = $out }
    [IO.File]::WriteAllText("D:\tb-gpu-runs\_spec_yard_tw1.json", ($s | ConvertTo-Json -Depth 5)); & D:\tb-gpu-runs\tools\make_sheet.ps1 -SpecPath "D:\tb-gpu-runs\_spec_yard_tw1.json" | Out-Null; return $out
}
$generic = @("Placeholder or untextured materials (flat grey, checker or faceted blocks)", "Floating bodies or props (a gap between an object and the ground under it)", "Clipping (geometry passing through geometry, or the camera inside geometry)")
$p2011 = "P2-011: named destinations read as unmarked grassy slopes or small generic structures instead of distinct halls, arches or terraces"
$names = @{ "pumps-shore" = "the pumps shore on Sluice Isle"; "jetty" = "the jetty below Shellwatch" }
$sets = @(); $cover = @()
foreach ($pr in "Medium", "High") {
    foreach ($v in "pumps-shore", "jetty") {
        $fa = "$o\TW1-cand-$pr\frames\$v.jpg"; $fb = "$o\TW1-ship-$pr\frames\$v.jpg"
        if (-not ((Test-Path $fa) -and (Test-Path $fb))) { "MISSING $v $pr"; continue }
        $sa = Side @($fa) @($v) "$od\TW1-$v-cand-$pr.png"; $sb = Side @($fb) @($v) "$od\TW1-$v-ship-$pr.png"
        $brief = "One fixed production-camera view of a named Tidewake destination ($($names[$v])) at noon, with the trainer as a 1.80 m ruler and the game HUD on screen. Judge whether the destination reads as a distinct, memorable place in the key art's Tidewake - open sea and shore, a landmark you could navigate by, believable construction that sits in the world. Ignore the HUD, and ignore which wild creatures happen to be on the beach (they spawn differently each capture)."
        $sets += [ordered]@{ label = "TW1 $v $pr cand|ship"; a = $sa; b = $sb; brief = $brief; defects = @($p2011) + $generic }
        $cover += [ordered]@{ view = $v; preset = $pr }
    }
}
$rng = New-Object System.Random 2141; $sets = @($sets | Sort-Object { $rng.Next() })
New-Item -ItemType Directory -Force "D:\tb-gpu-runs\yard\TW1\b1" | Out-Null; [IO.File]::WriteAllText("D:\tb-gpu-runs\yard\TW1\b1\candidates.json", (ConvertTo-Json -InputObject @($sets) -Depth 5))
[IO.File]::WriteAllText("D:\tb-gpu-runs\yard\TW1\coverage.json", (ConvertTo-Json -InputObject @($cover) -Depth 4))
"TW1 sets: $($sets.Count)"
