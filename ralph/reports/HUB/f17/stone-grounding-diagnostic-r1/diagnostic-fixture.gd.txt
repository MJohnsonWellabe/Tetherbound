extends SceneTree

## Regression for CI6200's wood-to-stone walk on the actual regional bake.
## Declared post-opening starter and one initial pose from the failed receipt.
## Then ordinary parsed controller input, native terrain/capsule and gravity.
## This diagnostic never claims the continuous earned M1 chain.
const OPENING_FLAGS := ["opening:beat:wake", "opening:beat:house", "opening:beat:choose",
	"opening:starter_granted", "opening:beat:name", "opening:beat:return_starter", "opening:beat:walk_out"]
const NAV := preload("res://tests/helpers/stick_navigator.gd")
var samples: Array[Dictionary] = []

func _init() -> void:
	_run.call_deferred()

func _run() -> void:
	var game := root.get_node("Game")
	for flag: String in OPENING_FLAGS:
		game.get("progression").call("set_flag", flag)
	game.get("party").call("add", game.call("make_creature", "terrapup"))
	var world := (load("res://scenes/world/meadows_playground.tscn") as PackedScene).instantiate()
	root.add_child(world)
	current_scene = world
	for frame in 180:
		await physics_frame
	var player := world.get_node("Player") as CharacterBody3D
	player.set_meta("f17_trace", true)
	var rig := world.get_node("CameraRig") as Node3D
	player.global_position = Vector3(29.0068206787109, .192472666501999, -24.138053894043)
	player.velocity = Vector3.ZERO
	for frame in 30:
		await physics_frame
	var navigation: RefCounted = NAV.new(self, player, rig, _drive)
	var target := Vector3(22, .76475590467453, -34)
	var airborne := 0
	for frame in 900:
		navigation.step(target)
		await physics_frame
		if not player.is_on_floor():
			airborne += 1
		var contacts: Array[String] = []
		for index in player.get_slide_collision_count():
			var collider := player.get_slide_collision(index).get_collider() as Node
			contacts.append(str(collider.get_path()) if collider != null else "non-node")
		var sample := {"frame": frame, "body": [player.global_position.x, player.global_position.y, player.global_position.z],
			"on_floor": player.is_on_floor(), "contacts": contacts}
		if not player.is_on_floor():
			var params := PhysicsTestMotionParameters3D.new()
			params.from = player.global_transform
			params.motion = Vector3.DOWN * player.floor_snap_length
			params.max_collisions = 4
			var result := PhysicsTestMotionResult3D.new()
			var hit := PhysicsServer3D.body_test_motion(player.get_rid(), params, result)
			var sweep: Array = []
			for index in result.get_collision_count():
				var collider := result.get_collider(index) as Node
				sweep.append({"collider": str(collider.get_path()) if collider != null else "non-node", "normal": str(result.get_collision_normal(index))})
			sample["sweep"] = {"hit": hit, "travel": str(result.get_travel()), "contacts": sweep}
			sample["slide"] = str(player.get_meta("f17_slide"))
			sample["step"] = str(player.get_meta("f17_step"))
		samples.append(sample)
		if Vector2(player.global_position.x - target.x, player.global_position.z - target.z).length() < 1.55:
			break
	_drive(0, 0)
	var reached := Vector2(player.global_position.x - target.x, player.global_position.z - target.z).length() < 1.55
	print("F17_STONE_GROUNDING ", JSON.stringify({"airborne_frames": airborne, "reached": reached,
		"samples": samples, "fixture": "post-opening flags/starter, one CI6200 initial pose; parsed sticks thereafter"}))
	quit(0 if airborne == 0 and reached else 1)

func _drive(x: float, y: float) -> void:
	for pair: Array in [[JOY_AXIS_LEFT_X, x], [JOY_AXIS_LEFT_Y, y]]:
		var event := InputEventJoypadMotion.new()
		event.device = 0
		event.axis = pair[0]
		event.axis_value = pair[1]
		Input.parse_input_event(event)
	Input.flush_buffered_events()
