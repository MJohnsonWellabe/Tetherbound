extends SceneTree
var frames := 0
var path := "res://scenes/world/cloudreach_cliffs.tscn"
var t: Thread
var got := false
func _load() -> Resource:
	return ResourceLoader.load(path)
func _process(_d: float) -> bool:
	frames += 1
	if frames == 3: change_scene_to_file("res://scenes/world/meadows_playground.tscn")
	if frames == 100:
		t = Thread.new(); t.start(_load)
	if frames > 100 and not got and t != null and not t.is_alive():
		var r = t.wait_to_finish(); got = true; print("TMP_GOT ", r, " at ", frames)
	return got and frames > 400 or frames > 20000
