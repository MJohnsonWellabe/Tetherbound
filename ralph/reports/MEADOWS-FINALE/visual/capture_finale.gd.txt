extends SceneTree

## F05 visual evidence: what a player sees at the Meadows finale, through the
## PRODUCTION player camera rig (the scene's own CameraRig following the real
## Player), with the HUD on. Nothing is hidden and nothing is touched up.
##
##   xvfb-run -a -s "-screen 0 1280x720x24" godot --path . --rendering-driver opengl3 \
##     --resolution 1280x720 --script <this file> -- <out_dir>
##
## Disclosed staging: `defeated_warden` is set directly (no Warden fight), the
## party is built with `make_creature`, and the player is placed at each
## vantage and turned to face its subject; the rig's yaw is set behind the
## player the way `camera_recenter` sets it. The lever, the dialogue and the
## prompts are the real ones, pressed with injected `interact`.

const SCENE := "res://scenes/world/meadows_playground.tscn"
const SETTLE := 120
const SHOT_FRAMES := 24

var _out := "user://f05_finale_shots"
var _world: Node3D
var _game: Node
var _player: CharacterBody3D
var _rig: Node3D
var _climax: Node


func _init() -> void:
	_run.call_deferred()


func _run() -> void:
	var args := OS.get_cmdline_user_args()
	if args.size() > 0:
		_out = args[0]
	DirAccess.make_dir_recursive_absolute(_out)
	await process_frame
	# Software GL renders every frame whether or not it is saved. Rendering is
	# switched off between captures and on only for the frames that are kept;
	# the simulation still runs every frame. What is saved is still the real
	# production camera's frame.
	RenderingServer.render_loop_enabled = false
	_game = root.get_node(^"Game")
	var party: RefCounted = _game.get("party")
	party.call("clear")
	for species: String in ["terrapup", "mudsnout", "bramblebun", "brooktail"]:
		party.call("add", _game.call("make_creature", species, species.capitalize()))
	# The Warden's own reward grants the Cloudreach key; no fight is played
	# here, so both of its durable results are set directly (disclosed).
	for flag in ["defeated_warden", "realm_key_cloudreach"]:
		_game.get("progression").call("set_flag", flag)
	_world = (load(SCENE) as PackedScene).instantiate()
	root.add_child(_world)
	current_scene = _world
	for i in SETTLE:
		await process_frame
	_player = _world.get_node(^"Player") as CharacterBody3D
	_rig = get_first_node_in_group("camera_rig") as Node3D
	_climax = _world.get_node(^"StrongholdClimax")

	# BEFORE: two real drain stations (terrain_playground.json drains.stations):
	# the quarry conduit head and the Hall approach run with its pylons.
	await _shot("01-quarry-before-healing", Vector2(378.0, 1832.0), Vector3(404.0, 0.0, 1804.0))
	await _shot("01b-hall-approach-before-healing", Vector2(-46.0, 7190.0), Vector3(-26.3, 0.0, 7222.1))

	# The lever, from the machine control, facing the bound creature.
	var control := _climax.find_child("MachineControl", true, false) as Node3D
	var bound := _climax.call("legendary_body") as Node3D
	await _shot_at("02-lever-prompt", control.global_position, bound.global_position)
	await _press("interact")
	# Read through the chamber and the join offer until the choice opens.
	var panel := _world.get_node_or_null(^"DialoguePanel")
	var saw_dialogue := false
	for i in 1500:
		await process_frame
		if bool(_climax.call("choice_open")):
			break
		if panel != null and bool(panel.call("is_open")):
			if not saw_dialogue:
				saw_dialogue = true
				await _grab("03-chamber-dialogue")
			await _press("interact")
	var stag := _climax.call("legendary_body") as Node3D
	# The announcement is said announce_delay after the offer opens; save the
	# frame the moment the HUD strip shows it (its life is 2.2 s wall-clock).
	var line := _world.find_child("Message", true, false) as Label
	for i in 600:
		await process_frame
		if line != null and line.is_visible_in_tree() and line.text.contains("final"):
			break
	await _grab("04-offer-open-announce")
	var accept := _world.find_child("VeridianAcceptPrompt", true, false) as Node3D
	var refuse := _world.find_child("VeridianRefusePrompt", true, false) as Node3D
	await _shot_at("05-accept-prompt", accept.get_parent().global_position, stag.global_position)
	await _shot_at("06-refuse-prompt", refuse.get_parent().global_position, stag.global_position)
	await _press("interact")  # refuse, standing at the refuse prompt
	for i in 20:
		await process_frame
	await _grab("06b-refused-line")
	for i in 90:
		await process_frame
		if panel != null and bool(panel.call("is_open")):
			await _press("interact")
	print("answered: joined=%s refused=%s" % [
		str(_game.get("progression").call("has", "legendary_joined")),
		str(_game.get("progression").call("has", "legendary_refused"))])
	# Let the healing fades run (dead-ground skin fades over 12 s).
	for i in 60 * 14:
		await process_frame

	# AFTER: the same quarry vantage, healed.
	await _shot("07-quarry-after-healing", Vector2(378.0, 1832.0), Vector3(404.0, 0.0, 1804.0))
	await _shot("07b-hall-approach-after-healing", Vector2(-46.0, 7190.0), Vector3(-26.3, 0.0, 7222.1))
	# The full-refusal herd display with the Highfield herd.
	var healing := _world.get_node(^"MeadowHealing")
	var herd := healing.call("herd_display") as Node3D
	print("herd display: %s" % (str(herd.global_position) if herd != null else "ABSENT"))
	if herd != null:
		var at := herd.global_position
		# The Highfield's authored hero stand (tools/capture_highfield_hero_identity.gd).
		await _shot("08-herd-display-hero-stand", Vector2(400.0, 5832.0), at)
		await _shot("09-herd-display-hero-framing", Vector2(400.0, 5832.0), Vector3(408.0, 0.0, 5870.0))
		await _shot("10-herd-display-close", Vector2(at.x + 6.0, at.z - 12.0), at)
	print("done herd")
	await _crossing_sequence()
	print("done: %s" % ProjectSettings.globalize_path(_out))
	quit(0)


## Stand at `flat` on the ground, face `target`, recentre the rig, capture.
func _shot(name: String, flat: Vector2, target: Vector3) -> void:
	var y := float(_world.call("ground_height_at", flat.x, flat.y))
	var t := target
	if t.y == 0.0:
		t.y = float(_world.call("ground_height_at", t.x, t.z))
	await _shot_at(name, Vector3(flat.x, y, flat.y), t)


func _shot_at(name: String, at: Vector3, target: Vector3) -> void:
	_player.global_position = at + Vector3(0.0, 0.3, 0.0)
	_player.velocity = Vector3.ZERO
	var dir := target - at
	dir.y = 0.0
	if dir.length() > 0.01:
		_player.rotation.y = atan2(dir.x, dir.z)
		if _rig != null:
			_rig.set("yaw", wrapf(atan2(-dir.x, -dir.z), -PI, PI))
			_rig.set("pitch", deg_to_rad(-8.0))
	var started := Time.get_ticks_msec()
	for i in SHOT_FRAMES:
		await process_frame
	print("  %s: %d frames in %.1f s" % [name, SHOT_FRAMES, (Time.get_ticks_msec() - started) / 1000.0])
	await _grab(name)


func _grab(name: String) -> void:
	var path := "%s/%s.png" % [_out, name]
	# One synchronous draw in the SAME frame the image is read, so a HUD line
	# with a 2.2 s wall-clock life is still up when the frame is saved (a
	# software-GL frame costs seconds; waiting several frames outlived it).
	RenderingServer.force_draw(false)
	root.get_texture().get_image().save_png(path)
	var hud_line := _world.find_child("Message", true, false) as Label if is_instance_valid(_world) else null
	print("shot -> %s   [hud message visible=%s text='%s']" % [path,
		str(hud_line != null and hud_line.is_visible_in_tree()), hud_line.text if hud_line != null else ""])


func _press(action: String) -> void:
	for pressed: bool in [true, false]:
		var event := InputEventAction.new()
		event.action = action
		event.pressed = pressed
		Input.parse_input_event(event)
		if pressed:
			Input.action_press(action)
		else:
			Input.action_release(action)
		for i in 3:
			await process_frame


## The physical gate crossing, as ordinary input: the player stands on the
## storm road short of the rebuilt span and holds forward (`move_forward`) through
## it, one kept frame per second of play for up to 30 s or until the realm
## changes. The production camera follows the whole way.
func _crossing_sequence() -> void:
	var crossing := _world.get_node_or_null(^"RiftCrossing")
	if crossing == null or not bool(crossing.call("span_ready")):
		print("crossing: no ready span; skipped")
		return
	var start: Vector3 = crossing.call("near_anchor")
	var far: Vector3 = crossing.call("far_anchor")
	var trigger := crossing.find_child("RiftCrossingTrigger", true, false) as Node3D
	var goal: Vector3 = trigger.global_position if trigger != null else far
	# At least 30 s of ordinary walking: start back down the storm road on the
	# span's own axis, so the approach, the span and the far rim are all in it.
	var axis := start - goal
	axis.y = 0.0
	var from := start + axis.normalized() * float(OS.get_environment("F05_CROSSING_BACK_M") if OS.get_environment("F05_CROSSING_BACK_M") != "" else "60")
	from.y = float(_world.call("ground_height_at", from.x, from.z))
	await _shot_at("20-crossing-approach", from, goal)
	Input.action_press("move_forward")
	var realm := str(_game.get("current_realm"))
	var arrived_at := -1
	# 30 kept frames, one per second of play, the whole way: the approach, the
	# span, the realm change, and walking on into Cloudreach with forward still
	# held. After the realm change the new scene's player and rig are followed.
	for second in 30:
		for i in 60:
			await process_frame
			if arrived_at < 0 and str(_game.get("current_realm")) != realm:
				arrived_at = second
				print("crossing: realm changed to '%s' during second %d" % [str(_game.get("current_realm")), second + 1])
			if arrived_at < 0 and i % 10 == 0 and _rig != null and is_instance_valid(_rig) and is_instance_valid(_player):
				var to := goal - _player.global_position
				to.y = 0.0
				if to.length() > 0.1:
					_rig.set("yaw", wrapf(atan2(-to.x, -to.z), -PI, PI))
		if arrived_at >= 0:
			# The new scene's HUD is what is on screen now.
			_world = current_scene as Node3D
		await _grab("2%02d-crossing-t%02ds" % [1 + second, second + 1])
	Input.action_release("move_forward")
	var party: RefCounted = _game.get("party")
	print("crossing: realm now '%s', scene '%s', party %d" % [str(_game.get("current_realm")),
		str(current_scene.name) if current_scene != null else "", int(party.call("size")) if party.has_method("size") else -1])
