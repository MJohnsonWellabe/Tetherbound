extends SceneTree

const CREATURE = preload("res://scenes/creatures/creature.tscn")
const BODY = preload("res://scripts/creatures/creature_body.gd")
const OUT = "D:/tetherbound/visual-acceptance-local/galecrest-look-directions"

func _init():
	_run.call_deferred()

func _run():
	await process_frame
	root.size = Vector2i(1920,1080)
	var world = Node3D.new()
	root.add_child(world)
	var env = WorldEnvironment.new()
	env.environment = Environment.new()
	env.environment.background_mode = Environment.BG_COLOR
	env.environment.background_color = Color(0.24,0.28,0.32)
	env.environment.ambient_light_source = Environment.AMBIENT_SOURCE_COLOR
	env.environment.ambient_light_color = Color(0.8,0.85,1.0)
	env.environment.ambient_light_energy = 0.65
	world.add_child(env)
	var light = DirectionalLight3D.new()
	light.rotation_degrees = Vector3(-40,-30,0)
	light.light_energy = 1.0
	world.add_child(light)
	var cam = Camera3D.new()
	world.add_child(cam)
	cam.position = Vector3(3,2.8,5)
	cam.look_at(Vector3(0,1.8,0))
	cam.make_current()
	DirAccess.make_dir_recursive_absolute(OUT)
	var body = CREATURE.instantiate()
	body.set_script(BODY)
	world.add_child(body)
	body.setup("galecrest")
	body.set_physics_process(false)
	body.set_process(false)
	await process_frame
	var player = body.find_children("*","AnimationPlayer",true,false)[0] as AnimationPlayer
	player.callback_mode_process = AnimationMixer.ANIMATION_CALLBACK_MODE_PROCESS_MANUAL
	player.play("idle")
	player.seek(1.2,true)
	player.advance(0)
	var leader = Node3D.new()
	world.add_child(leader)
	leader.position = Vector3(3,0,1.5)
	var presence = load("res://scripts/creatures/companion_presence.gd").new()
	body.add_child(presence)
	presence.setup(body)
	presence._build_look(presence.get("_cfg").get("look",{}),leader)
	var look = presence.get("_look") as LookAtModifier3D
	print("LOOK forward=",look.forward_axis," primary=",look.primary_rotation_axis," relative=",look.relative)
	for view in ["front"]:
		cam.position = Vector3(0,2.8,6) if view == "front" else Vector3(5,2.8,0)
		cam.look_at(Vector3(0,1.8,0))
		for variant in ["original_left", "candidate_left", "original_right", "candidate_right", "original_rear", "candidate_rear"]:
			look.active = true
			leader.position = Vector3(-5,0,1) if "left" in variant else Vector3(5,0,1)
			if "rear" in variant: leader.position = Vector3(1,0,-5)
			if "candidate" in variant: leader.position.y = 1.55
			look.primary_limit_angle = deg_to_rad(55 if "candidate" in variant else 110)
			look.secondary_limit_angle = deg_to_rad(30 if "candidate" in variant else 60)
			look.primary_rotation_axis = Vector3.AXIS_Z if "candidate" in variant else Vector3.AXIS_Y
			for tick in 45: await process_frame
			await shoot(view+"-"+variant)

	quit()

func shoot(label):
	await process_frame
	await RenderingServer.frame_post_draw
	var img = root.get_texture().get_image()
	img.save_png(OUT+"/"+label+".png")
	print(label," ",img.get_size())
