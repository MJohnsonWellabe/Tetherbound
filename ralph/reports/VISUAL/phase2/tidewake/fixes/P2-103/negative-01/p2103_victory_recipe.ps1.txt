function Get-CaptureCases([string]$RequestedSize, [bool]$OnlyNegative) {
    $resolutions = @(if ($RequestedSize -eq 'both') { '1080'; '720' } else { $RequestedSize })
    $cases = @([ordered]@{ id='venn-cap-negative'; trainer='venn'; size=$resolutions[0]; variant='before'; kind='negative_frame_cap' })
    if (-not $OnlyNegative) {
        foreach ($resolution in $resolutions) {
            foreach ($trainer in @('venn','nerissa')) {
                foreach ($variant in @('before','after')) {
                    $cases += [ordered]@{ id="$trainer-$resolution-$variant"; trainer=$trainer; size=$resolution; variant=$variant; kind='comparison' }
                }
            }
        }
    }
    return $cases
}

function Read-RenderLeaseState([IO.Stream]$Stream) {
    $Stream.Position = 0
    $reader = [IO.StreamReader]::new($Stream, [Text.Encoding]::UTF8, $true, 1024, $true)
    try { $text = $reader.ReadToEnd() } finally { $reader.Dispose() }
    if ([string]::IsNullOrWhiteSpace($text)) { return @{} }
    $state = $text | ConvertFrom-Json
    if ($null -eq $state -or $state -isnot [pscustomobject]) { throw 'Render lock has an invalid state; no claim made.' }
    return $state
}

function Claim-RenderLeaseStream([IO.Stream]$Stream, [string]$Token, [string]$Purpose) {
    $state = Read-RenderLeaseState $Stream
    if ($state.held_by -or $state.token -or $state.run_token) { throw 'Render lock is occupied; no queue slot claimed.' }
    $bytes = [Text.Encoding]::UTF8.GetBytes((@{ held_by='tb/x04-tidewake'; token=$Token; run_token=$Token;
        since_utc=[DateTime]::UtcNow.ToString('o'); purpose=$Purpose; launcher_pid=$PID } | ConvertTo-Json -Compress))
    $Stream.Position = 0
    $Stream.SetLength(0)
    $Stream.Write($bytes, 0, $bytes.Length)
    $Stream.Flush()
}

function Open-RenderLeaseStream([string]$Path) {
    return [IO.File]::Open($Path, [IO.FileMode]::OpenOrCreate, [IO.FileAccess]::ReadWrite, [IO.FileShare]::None)
}

function Enter-RenderLease([string]$Path, [string]$Token, [string]$Purpose) {
    $handle = Open-RenderLeaseStream $Path
    try {
        Claim-RenderLeaseStream $handle $Token $Purpose
        # Keep FileShare.None through the whole batch, including child cleanup.
        return [pscustomobject]@{ Stream=$handle; Token=$Token; Path=$Path }
    } catch {
        try { $handle.Dispose() } catch { Write-Warning "Render lock handle cleanup failed: $($_.Exception.Message)" }
        throw
    }
}

function Release-RenderLease($Lease, [bool]$ChildTerminal) {
    if (-not $ChildTerminal) { throw 'Owned child is not verified terminal; render token retained.' }
    $state = Read-RenderLeaseState $Lease.Stream
    if ($state.token -cne $Lease.Token -or $state.run_token -cne $Lease.Token -or $state.held_by -cne 'tb/x04-tidewake') {
        throw 'Render lock ownership changed; no token cleared.'
    }
    $bytes = [Text.Encoding]::UTF8.GetBytes('{}')
    $Lease.Stream.Position = 0
    $Lease.Stream.SetLength(0)
    $Lease.Stream.Write($bytes, 0, $bytes.Length)
    $Lease.Stream.Flush()
}

function Wait-CaptureChild($Process, [int]$TimeoutMilliseconds = 1800000) {
    $timer = [Diagnostics.Stopwatch]::StartNew()
    try {
        while ($true) {
            $remaining = $TimeoutMilliseconds - [int]$timer.ElapsedMilliseconds
            if ($remaining -le 0) { throw [TimeoutException]::new('Owned capture exceeded its 30-minute process deadline.') }
            if ($Process.WaitForExit([Math]::Min(1000, $remaining))) {
                $Process.Refresh()
                if (-not $Process.HasExited) { throw 'Child wait completed without a verified terminal process.' }
                return
            }
        }
    } finally { $timer.Stop() }
}

function Stop-OwnedCaptureChild($Process) {
    # Only the Process object returned by our Start-Process, never a PID/name search.
    Stop-Process -InputObject $Process -Force -ErrorAction Stop
}

function Resolve-OwnedCaptureChild($Process) {
    if ($null -eq $Process) { return [pscustomobject]@{ Terminal=$true; Error='' } }
    try {
        $Process.Refresh()
        if (-not $Process.HasExited) {
            try { Stop-OwnedCaptureChild $Process } catch {
                $stopError = $_.Exception.Message
                # A natural exit between HasExited and Stop-Process is successful cleanup.
                $Process.Refresh()
                if (-not $Process.HasExited) { return [pscustomobject]@{ Terminal=$false; Error=$stopError } }
            }
            if (-not $Process.WaitForExit(10000)) {
                return [pscustomobject]@{ Terminal=$false; Error='Owned child did not terminate within the 10-second cleanup bound.' }
            }
        }
        $Process.Refresh()
        if (-not $Process.HasExited) { return [pscustomobject]@{ Terminal=$false; Error='Owned child terminal state cannot be verified.' } }
        return [pscustomobject]@{ Terminal=$true; Error='' }
    } catch { return [pscustomobject]@{ Terminal=$false; Error=$_.Exception.Message } }
}

function Get-LfHash([string]$Path) {
    $sha = [Security.Cryptography.SHA256]::Create()
    try {
        $bytes = [Text.Encoding]::UTF8.GetBytes([IO.File]::ReadAllText($Path).Replace("`r`n", "`n"))
        return ([BitConverter]::ToString($sha.ComputeHash($bytes))).Replace('-', '').ToLowerInvariant()
    } finally { $sha.Dispose() }
}

            'Input.action_press("interact")', 'Input.action_release("interact")',
            'Post-win dialogue requires a decision', 'int(event.get("seq", 0)) > after_seq',
            'str(state.get("banner", "")).contains(text)', 'await RenderingServer.frame_post_draw',
            'current.seq) == int(receipt.seq)', 'phase == "aftermath" and elapsed_ms < 1000',
            'image.get_size() != _expected_resolution', 'if image.save_png(path) != OK:',
            'func lifecycle_manifest(', '"missing_phases": missing',
            'func _finalize_capture(failures: int)', 'Inherited capture/raw frame export failed')

    }
    return [ordered]@{ complete=(@($checks | Where-Object { -not $_.hash_matches -or $_.missing_anchors.Count -gt 0 }).Count -eq 0);
        review_base_commit=$metadata.base_commit; checks=$checks }
}

function Assert-OverridePack([string]$Path, [string]$OriginalJson, [bool]$Enabled) {
    $archive = [IO.Compression.ZipFile]::OpenRead($Path)
    try {
        if ($archive.Entries.Count -ne 1 -or $archive.Entries[0].FullName -cne 'data/config/victory_hierarchy_visual.json') {
            throw 'Override pack scope is not exactly the victory config.'
        }
        $reader = [IO.StreamReader]::new($archive.Entries[0].Open())
        try { $packed = $reader.ReadToEnd() | ConvertFrom-Json } finally { $reader.Dispose() }
        $expected = $OriginalJson | ConvertFrom-Json
        $expected.enabled = $Enabled
        if (($packed | ConvertTo-Json -Depth 30 -Compress) -cne ($expected | ConvertTo-Json -Depth 30 -Compress)) {
            throw 'Override config differs from current production config beyond enabled.'
        }
    } finally { $archive.Dispose() }
}

function Assert-CaptureResult($Plan, [int]$ExitCode, [string]$Project, [string]$Logs) {
    $directory = Join-Path $Project $Plan.output.Substring(6)
    $manifest = Get-Content -LiteralPath (Join-Path $directory 'manifest.json') -Raw | ConvertFrom-Json
    if ($manifest.attempt.source_commit -cne $Plan.source_head -or
        $manifest.attempt.override_pack_sha256 -ine $Plan.override_zip_sha256 -or
        $manifest.attempt.victory_config.enabled -ne ($Plan.variant -eq 'after')) { throw 'Manifest source/pack/gate identity mismatch.' }
    $errors = @($Logs -split "`n" | Where-Object { $_ -match '^\s*(SCRIPT ERROR|ERROR):|Parse Error|Parser Error' })
    if ($Plan.kind -eq 'negative_frame_cap') {
        $unexpected = @($errors | Where-Object { $_ -notmatch '^\s*ERROR: Capture rejected: frame_cap\s*$' })
        if ($ExitCode -ne 1 -or $manifest.complete -ne $false -or
            $manifest.attempt.terminal.reason -cne 'frame_cap' -or $manifest.attempt.terminal.ok -ne $false -or
            ($manifest.missing_phases -join '|') -cne 'win|aftermath|receipt-expired' -or
            $manifest.attempt.defeat_before -ne $false -or $manifest.attempt.defeat_after -ne $false -or
            @($manifest.frames | Where-Object { $_.phase -in @('win','aftermath','receipt-expired') }).Count -gt 0 -or
            @($manifest.frames | Where-Object { $_.phase -eq 'fight-observation' }).Count -eq 0 -or $unexpected.Count -gt 0) {
            throw 'Negative control did not fail specifically and cleanly at frame_cap; stop before comparison.'
        }
    } else {
        if ($ExitCode -ne 0 -or $errors.Count -gt 0 -or $manifest.complete -ne $true -or
            @($manifest.failures).Count -gt 0 -or @($manifest.missing_phases).Count -gt 0 -or
            $manifest.attempt.terminal.ok -ne $true -or $manifest.attempt.defeat_before -ne $false -or
            $manifest.attempt.defeat_after -ne $true) { throw 'Capture failed its verified lifecycle or engine error checks.' }
        foreach ($phase in @('win','aftermath','receipt-expired')) {
            $matching = @($manifest.frames | Where-Object { $_.phase -eq $phase })
            if ($matching.Count -ne 1) { throw "Expected one $phase frame." }
            $frame = $matching[0]
            if (($frame.resolution -join 'x') -cne $Plan.resolution) { throw 'Lifecycle image metadata has wrong resolution.' }
            $imagePath = Join-Path $Project $frame.file.Substring(6)
            $stream = [IO.File]::OpenRead($imagePath)
            try { $header = [byte[]]::new(24); $read = $stream.Read($header, 0, 24) } finally { $stream.Dispose() }
            if ($read -ne 24 -or [BitConverter]::ToString($header, 0, 8) -cne '89-50-4E-47-0D-0A-1A-0A') { throw 'Lifecycle image is not a PNG.' }
            $width = [Net.IPAddress]::NetworkToHostOrder([BitConverter]::ToInt32($header, 16))
            $height = [Net.IPAddress]::NetworkToHostOrder([BitConverter]::ToInt32($header, 20))
            if ("${width}x${height}" -cne $Plan.resolution) { throw 'Actual PNG dimensions do not match the requested native raster.' }
            if ($phase -eq 'aftermath' -and $frame.elapsed_since_win_ms -lt 1000) { throw 'Aftermath was captured before one real second.' }
        }
    }
    return [ordered]@{ id=$Plan.id; kind=$Plan.kind; expected_exit=$Plan.expected_exit; actual_exit=$ExitCode;
        checked=$true; native_visual_acceptance=$false; manifest_sha256=(Get-FileHash -LiteralPath (Join-Path $directory 'manifest.json') -Algorithm SHA256).Hash }
}
